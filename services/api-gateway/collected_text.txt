# app/middleware/jwt_validation.py
import jwt
from fastapi import Request
from fastapi.responses import JSONResponse
from pathlib import Path
from starlette.middleware.base import BaseHTTPMiddleware

from app.config import settings


PUBLIC_ROUTES = {
    "/auth/login",
    "/auth/register",
    "/auth/verify-otp",
    "/auth/refresh",
    "/auth/health",
    "/api/auth/login",
    "/api/auth/register",
    "/api/auth/verify-otp",
    "/api/auth/refresh",
    "/api/auth/health",
    "/docs",
    "/openapi.json",
    "/redoc",
    "/health",
}


def _is_public(path: str) -> bool:
    # برابری دقیق یا زیرمسیر واقعی (با "/" بعدش)، نه فقط startswith خام
    # تا مثلاً "/auth/login-something" به اشتباه عمومی تشخیص داده نشود.
    if path in PUBLIC_ROUTES:
        return True
    return any(path.startswith(route + "/") for route in PUBLIC_ROUTES)


class JWTValidationMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, **kwargs):
        super().__init__(app)
        self.public_key = Path(settings.PUBLIC_KEY_PATH).read_text()

    async def dispatch(self, request: Request, call_next):
        path = request.url.path

        if request.method == "OPTIONS":
            return await call_next(request)

        if _is_public(path):
            return await call_next(request)

        auth_header = request.headers.get("authorization")
        if not auth_header:
            return JSONResponse(
                status_code=401,
                content={"detail": "Token not sent.", "code": "MISSING_TOKEN"}
            )

        if not auth_header.startswith("Bearer "):
            return JSONResponse(
                status_code=401,
                content={"detail": "The token format is invalid.", "code": "INVALID_TOKEN_FORMAT"}
            )

        token = auth_header.replace("Bearer ", "")

        try:
            payload = jwt.decode(
                token,
                self.public_key,
                algorithms=["RS256"],
                options={"require": ["exp", "sub", "role"]}
            )

            request.state.user_id = payload.get("sub")
            request.state.user_role = payload.get("role")
            request.state.token_payload = payload

        except jwt.ExpiredSignatureError:
            return JSONResponse(
                status_code=401,
                content={"detail": "Token has expired.", "code": "TOKEN_EXPIRED"}
            )
        except jwt.InvalidTokenError as e:
            return JSONResponse(
                status_code=401,
                content={"detail": f"Token is invalid: {str(e)}", "code": "INVALID_TOKEN"}
            )

        return await call_next(request)

import time
import json
from fastapi import Request
from starlette.middleware.base import BaseHTTPMiddleware


class LoggingMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        start_time = time.time()
        
        # اطلاعات درخواست
        method = request.method
        path = request.url.path
        query = str(request.query_params)
        client_ip = request.client.host if request.client else "unknown"
        user_agent = request.headers.get("user-agent", "unknown")
        user_id = getattr(request.state, "user_id", "anonymous")
        user_role = getattr(request.state, "user_role", "none")
        
        # ارسال درخواست
        try:
            response = await call_next(request)
            status_code = response.status_code
        except Exception as exc:
            status_code = 500
            raise exc
        finally:
            process_time = (time.time() - start_time) * 1000  # میلی‌ثانیه
            
            log_entry = {
                "timestamp": time.strftime("%Y-%m-%dT%H:%M:%S"),
                "method": method,
                "path": path,
                "query": query if query else None,
                "status_code": status_code,
                "process_time_ms": round(process_time, 2),
                "client_ip": client_ip,
                "user_id": user_id,
                "user_role": user_role,
                "user_agent": user_agent,
            }
            
            # لاگ در stdout (در Docker با json formatter قابل جمع‌آوری است)
            print(json.dumps(log_entry, ensure_ascii=False))
        
        # اضافه کردن هدر زمان پاسخ
        response.headers["X-Process-Time"] = str(round(process_time, 2))
        return response

# app/middleware/rate_limiter.py
import time
from fastapi import Request
from fastapi.responses import JSONResponse
from starlette.middleware.base import BaseHTTPMiddleware

from app.config import settings
from app.utils.redis_client import get_redis_client


class RateLimiterMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, **kwargs):
        super().__init__(app)
        self.redis = get_redis_client()
        self.limits = {
            "public": settings.RATE_LIMIT_PUBLIC,
            "auth": settings.RATE_LIMIT_AUTH,
            "profile": settings.RATE_LIMIT_PROFILE,
            "intake": settings.RATE_LIMIT_INTAKE,
            "records": settings.RATE_LIMIT_RECORD,
        }

    def _get_key(self, identifier: str, window: int) -> str:
        current_window = int(time.time()) // window
        return f"rate_limit:{identifier}:{current_window}"

    async def _check_limit(self, key: str, limit: int, window: int = 60) -> tuple[bool, int, int]:
        pipe = self.redis.pipeline()
        pipe.incr(key)
        pipe.expire(key, window)
        results = await pipe.execute()

        current_count = results[0]
        remaining = max(0, limit - current_count)
        reset_after = window - (int(time.time()) % window)

        allowed = current_count <= limit
        return allowed, remaining, reset_after

    async def _enforce(self, request, call_next, identifier: str, limit: int, error_code: str, detail: str):
        allowed, remaining, reset_after = await self._check_limit(self._get_key(identifier, 60), limit)
        if not allowed:
            return JSONResponse(
                status_code=429,
                content={"detail": detail, "code": error_code, "retry_after": reset_after},
                headers={"Retry-After": str(reset_after)},
            )
        response = await call_next(request)
        response.headers["X-RateLimit-Remaining"] = str(remaining)
        return response

    async def dispatch(self, request: Request, call_next):
        path = request.url.path

        # مسیرهای عمومی (لاگین/ثبت‌نام) — بر اساس IP، قبل از احراز هویت
        public_paths = ("/api/auth/login", "/api/auth/register", "/api/auth/verify-otp")
        if any(path.startswith(p) for p in public_paths):
            client_ip = request.client.host if request.client else "unknown"
            return await self._enforce(
                request, call_next,
                identifier=f"ip:{client_ip}",
                limit=self.limits["public"],
                error_code="RATE_LIMIT_EXCEEDED",
                detail="Access from this IP address has been rate-limited",
            )

        user_id = getattr(request.state, "user_id", None)

        # مسیرهای محافظت‌شده‌ای که به این‌جا می‌رسند باید user_id داشته باشند
        # (چون JWTValidationMiddleware قبل از این اجرا شده). اگر نداشت، یعنی مسیر
        # عمومی دیگری بوده که نیاز به rate limit خاص کاربر ندارد.
        if not user_id:
            return await call_next(request)

        if path.startswith("/api/profile/"):
            return await self._enforce(
                request, call_next,
                identifier=f"user:{user_id}:profile",
                limit=self.limits["profile"],
                error_code="PROFILE_RATE_LIMIT_EXCEEDED",
                detail="Profile service rate limit exceeded",
            )

        if path.startswith("/api/intake/"):
            return await self._enforce(
                request, call_next,
                identifier=f"user:{user_id}:intake",
                limit=self.limits["intake"],
                error_code="INTAKE_RATE_LIMIT_EXCEEDED",
                detail="Visit intake rate limit exceeded",
            )

        if path.startswith("/api/records/"):
            return await self._enforce(
                request, call_next,
                identifier=f"user:{user_id}:records",
                limit=self.limits["records"],
                error_code="RECORD_RATE_LIMIT_EXCEEDED",
                detail="Medical record rate limit exceeded",
            )

        # محدودیت عمومی برای هر کاربر احراز هویت‌شده روی سایر مسیرها
        return await self._enforce(
            request, call_next,
            identifier=f"user:{user_id}",
            limit=self.limits["auth"],
            error_code="RATE_LIMIT_EXCEEDED",
            detail="Rate limit exceeded for this user",
        )

# app/middleware/role_checker.py
from fastapi import Request
from fastapi.responses import JSONResponse
from starlette.middleware.base import BaseHTTPMiddleware


PROTECTED_ROUTES = {
    "/api/doctor-review/": ["DOCTOR", "ADMIN"],
    "/api/prescriptions/create": ["DOCTOR", "ADMIN"],
    "/api/prescriptions/approve": ["DOCTOR", "ADMIN"],
    "/api/admin/": ["ADMIN"],
    "/api/profile/patient/me": ["PATIENT"],
    "/api/profile/patient/me/allergies": ["PATIENT"],
    "/api/profile/patient/me/medications": ["PATIENT"],
    "/api/profile/patient/me/conditions": ["PATIENT"],
    "/api/profile/patient/me/avatar": ["PATIENT"],
    "/api/profile/doctor/me": ["DOCTOR"],
    "/api/profile/doctor/me/avatar": ["DOCTOR"],
}


class RoleCheckerMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, **kwargs):
        super().__init__(app)
    
    async def dispatch(self, request: Request, call_next):
        path = request.url.path
        user_role = getattr(request.state, "user_role", None)
        
        for route_prefix, allowed_roles in PROTECTED_ROUTES.items():
            if path.startswith(route_prefix):
                if user_role not in allowed_roles:
                    return JSONResponse(
                        status_code=403,
                        content={
                            "detail": "Forbidden: You do not have the required role to access this resource",
                            "code": "FORBIDDEN",
                            "required_roles": allowed_roles,
                            "your_role": user_role
                        }
                    )
                break
        
        return await call_next(request)

# app/router/proxy.py
import httpx
from fastapi import Request, APIRouter, HTTPException
from fastapi.responses import JSONResponse, Response

from app.config import settings


router = APIRouter()

SERVICES = {
    "auth": settings.AUTH_SERVICE_URL,
    "profile": settings.PROFILE_SERVICE_URL,
    "intake": settings.VISIT_INTAKE_SERVICE_URL,
    "records": settings.RECORD_SERVICE_URL,
}

# پیشوندی که هر سرویس واقعاً روت‌هایش را روش سوار کرده
ROUTE_PREFIX = {
    "auth": "/auth",       # رفتار فعلی حفظ می‌شود
    "profile": "/profile", # profile-service: /profile/patient, /profile/doctor
    "intake": "",          # visit-intake-service: /visits, /doctor/visits
    "records": "/records", # medical-record-service: /records
}

INTERNAL_ONLY = {"triage"}

# این هدرها نباید عیناً از پاسخ upstream به کلاینت کپی بشن؛ Starlette/FastAPI
# خودش باید content-length و content-encoding درست را برای بدنه‌ی جدید بسازد.
HOP_BY_HOP_HEADERS = {
    "connection", "keep-alive", "proxy-authenticate", "proxy-authorization",
    "te", "trailers", "transfer-encoding", "upgrade",
    "content-length", "content-encoding",
}


async def proxy_request(service: str, path: str, request: Request):
    """پروکسی درخواست به سرویس مقصد"""

    if service in INTERNAL_ONLY:
        raise HTTPException(status_code=403, detail="Forbidden: Internal service only")

    if service not in SERVICES:
        raise HTTPException(status_code=404, detail=f"Service '{service}' not found")

    # 🔒 مسیرهای internal نباید هرگز از Gateway عبور کنند
    if path.startswith("internal/"):
        raise HTTPException(status_code=403, detail="Internal endpoints are not accessible through the gateway")

    prefix = ROUTE_PREFIX.get(service, f"/api/{service}")
    target_url = f"{SERVICES[service]}{prefix}/{path}" if path else f"{SERVICES[service]}{prefix}"

    headers = dict(request.headers)
    headers.pop("host", None)

    user_id = getattr(request.state, "user_id", None)
    user_role = getattr(request.state, "user_role", None)

    if user_id:
        headers["x-user-id"] = str(user_id)
    if user_role:
        # سرویس‌ها (profile, visit-intake) با lowercase کار می‌کنند
        headers["x-user-role"] = str(user_role).lower()

    headers["x-gateway-request"] = "true"

    body = await request.body()

    async with httpx.AsyncClient(timeout=30.0) as client:
        try:
            response = await client.request(
                method=request.method,
                url=target_url,
                headers=headers,
                params=request.query_params,
                content=body if body else None,
            )
        except httpx.ConnectError:
            raise HTTPException(status_code=503, detail=f"Service '{service}' unavailable")
        except httpx.TimeoutException:
            raise HTTPException(status_code=504, detail=f"Timeout connecting to '{service}'")

    response_headers = {
        k: v for k, v in response.headers.items()
        if k.lower() not in HOP_BY_HOP_HEADERS
    }

    content_type = response.headers.get("content-type", "")

    # اگر پاسخ JSON است، به همان شکل قبلی برگردان.
    # اگر نیست (مثلاً تصویر آواتار یا فایل ضمیمه ویزیت)، بایت‌های خام را عیناً پاس بده
    # -- استفاده از response.json() روی محتوای غیر-JSON خطا می‌داد و درخواست را می‌شکست.
    if "application/json" in content_type:
        return JSONResponse(
            content=response.json() if response.content else {},
            status_code=response.status_code,
            headers=response_headers,
        )

    return Response(
        content=response.content,
        status_code=response.status_code,
        headers=response_headers,
        media_type=content_type or None,
    )

async def auth_proxy(path: str, request: Request):
