def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode."""
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def do_run_migrations(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata)

    with context.begin_transaction():
        context.run_migrations()


async def run_async_migrations() -> None:
    """In this scenario we need to create an Engine and associate a connection with the context."""
    connectable = async_engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    async with connectable.connect() as connection:
        await connection.run_sync(do_run_migrations)

    await connectable.dispose()


def run_migrations_online() -> None:
    """Run migrations in 'online' mode."""
    asyncio.run(run_async_migrations())


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

"""Authentication dependencies for FastAPI."""
from typing import Optional

from fastapi import Depends, HTTPException, status, Request
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import JWTError, jwt

from app.config import get_settings
from app.exceptions import InternalAPIKeyError

security = HTTPBearer(auto_error=False)


async def get_current_user(
    request: Request,
    credentials: Optional[HTTPAuthorizationCredentials] = Depends(security)
) -> dict:
    """Validate user identity from JWT OR trusted Gateway headers.
    
    Priority:
    1. If x-gateway-request header exists → Trust x-user-id/x-user-role (Gateway already validated)
    2. Otherwise → Validate JWT locally (direct access or other services)
    """
    settings = get_settings()
    
    # ➕ NEW: Check if request came from Gateway (trusted)
    is_gateway_request = request.headers.get("x-gateway-request") == "true"
    user_id_from_gateway = request.headers.get("x-user-id")
    user_role_from_gateway = request.headers.get("x-user-role")
    
    if is_gateway_request and user_id_from_gateway:
        # Gateway already validated JWT, trust the headers
        return {
            "sub": user_id_from_gateway,
            "role": user_role_from_gateway,
            "source": "gateway"
        }
    
    # ➕ Fallback: Validate JWT locally (direct access, testing, etc.)
    if not credentials:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authorization header missing",
            headers={"WWW-Authenticate": "Bearer"},
        )
    
    try:
        with open(settings.JWT_PUBLIC_KEY_PATH, "r") as f:
            public_key = f.read()
        
        payload = jwt.decode(
            credentials.credentials,
            public_key,
            algorithms=[settings.JWT_ALGORITHM],
        )
        
        user_id: str = payload.get("sub")
        if user_id is None:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid token: missing subject",
            )
        
        payload["source"] = "jwt"
        return payload
        
    except JWTError as e:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=f"Invalid token: {str(e)}",
            headers={"WWW-Authenticate": "Bearer"},
        )
    except FileNotFoundError:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="JWT public key not found",
        )


def require_role(required_role: str):
    """Dependency factory to enforce role-based access control."""
    async def role_checker(
        current_user: dict = Depends(get_current_user)
    ) -> dict:
        user_roles = current_user.get("roles", [])
        # Handle both list and single role
        if isinstance(user_roles, str):
            user_roles = [user_roles]
        user_role = current_user.get("role")
        if user_role:
            user_roles.append(user_role)
        
        if required_role not in user_roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Access denied. Required role: {required_role}",
            )
        return current_user
    
    return role_checker


async def verify_internal_api_key(
    request: Request,
    x_internal_api_key: Optional[str] = None
) -> bool:
    """Verify service-to-service API key for internal endpoints."""
    settings = get_settings()
    
    if x_internal_api_key is None:
        x_internal_api_key = request.headers.get("X-Internal-API-Key")
    
    if not x_internal_api_key or x_internal_api_key != settings.INTERNAL_API_KEY:
        raise InternalAPIKeyError()
    
    return True

"""FastAPI application factory for User Profile Service."""
from contextlib import asynccontextmanager

from fastapi import FastAPI, Request
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse

from app.config import get_settings
from app.routers import patient, doctor, internal

settings = get_settings()


@asynccontextmanager
async def lifespan(app: FastAPI):
    """Application lifespan handler for startup/shutdown events."""
    # Startup
    print(f"🚀 {settings.APP_NAME} starting up...")
    yield
    # Shutdown
    print(f"🛑 {settings.APP_NAME} shutting down...")


def create_application() -> FastAPI:
    """Create and configure the FastAPI application."""
    app = FastAPI(
        title=settings.APP_NAME,
        description="""
        Healthcare AI - User Profile Service
        
        Manages patient and doctor profiles including:
        - Basic demographic information
        - Allergies and reactions
        - Current medications
        - Chronic conditions
        - Profile avatars (stored in object storage)
        
        ## Authentication
        All patient/doctor endpoints require a valid JWT Bearer token.
        Internal endpoints require an `X-Internal-API-Key` header.
        
        ## HIPAA Compliance
        All access to Protected Health Information (PHI) is logged
        in the audit log for compliance purposes.
        """,
        version="1.0.0",
        lifespan=lifespan,
        docs_url="/docs" if settings.DEBUG else None,
        redoc_url="/redoc" if settings.DEBUG else None,
    )
    
    # CORS middleware
    app.add_middleware(
        CORSMiddleware,
        allow_origins=["*"] if settings.DEBUG else [],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )
    
    # Include routers
    app.include_router(patient.router)
    app.include_router(doctor.router)
    app.include_router(internal.router)
    
    # Global exception handler
    @app.exception_handler(Exception)
    async def global_exception_handler(request: Request, exc: Exception):
        """Handle uncaught exceptions gracefully."""
        return JSONResponse(
            status_code=500,
            content={"detail": "Internal server error"},
        )
    
    # Health check endpoint
    @app.get("/health", tags=["Health"])
    async def health_check():
        """Health check endpoint for load balancers and monitoring."""
        return {"status": "healthy", "service": settings.APP_NAME}
    
    # Root endpoint
    @app.get("/", tags=["Root"])
    async def root():
        """Root endpoint with service information."""
        return {
            "service": settings.APP_NAME,
            "version": "1.0.0",
            "docs": "/docs" if settings.DEBUG else None,
        }
    
    return app


# Create the application instance
app = create_application()

@router.post("/me/avatar")
async def upload_doctor_avatar(
    request: Request,
    file: UploadFile = File(...),
    current_user: dict = Depends(require_role("doctor")),
    db: AsyncSession = Depends(get_db),
):
    """Upload doctor profile picture."""
    storage = StorageService()
    service = ProfileService(db)
    audit = AuditService(db)
    
    user_id = UUID(current_user["sub"])
    
    image_url, object_key = await storage.upload_avatar(
        file=file,
        user_id=str(user_id),
        user_type="doctor",
    )
    
    old_profile = await service.get_doctor_profile(user_id)
    if old_profile and old_profile.profile_image_key:
        await storage.delete_file(old_profile.profile_image_key)
    
    # Note: need to add update_doctor_avatar to ProfileService
    # For now using direct update
    result = await db.execute(
        select(DoctorProfile).where(DoctorProfile.user_id == user_id)
    )
    profile = result.scalar_one_or_none()
    if profile:
        profile.profile_image_url = image_url
        profile.profile_image_key = object_key
        await db.flush()
    
    ip, ua = get_client_info(request)
    await audit.log_update(
        user_id=user_id,
        table_name="doctor_profiles",
        record_id=user_id,
        old_values={"profile_image_url": old_profile.profile_image_url if old_profile else None},
        new_values={"profile_image_url": image_url},
        performed_by=user_id,
        ip_address=ip,
        user_agent=ua,
    )
    
    return {"profile_image_url": image_url}

"""Authentication dependencies for FastAPI."""
from typing import Optional

from fastapi import Depends, HTTPException, status, Request
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import JWTError, jwt

from app.config import get_settings
from app.exceptions import InternalAPIKeyError

security = HTTPBearer(auto_error=False)


async def get_current_user(
    credentials: Optional[HTTPAuthorizationCredentials] = Depends(security)
) -> dict:
    """Validate JWT token and return user payload.
    
    This dependency is used for patient-facing endpoints.
    The JWT is signed by the Auth Service using RS256.
    """
    if not credentials:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authorization header missing",
            headers={"WWW-Authenticate": "Bearer"},
        )
    
    settings = get_settings()
    
    try:
        # Read public key from file
        with open(settings.JWT_PUBLIC_KEY_PATH, "r") as f:
            public_key = f.read()
        
        payload = jwt.decode(
            credentials.credentials,
            public_key,
            algorithms=[settings.JWT_ALGORITHM],
        )
        
        user_id: str = payload.get("sub")
        if user_id is None:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid token: missing subject",
            )
        
        return payload
        
    except JWTError as e:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=f"Invalid token: {str(e)}",
            headers={"WWW-Authenticate": "Bearer"},
        )
    except FileNotFoundError:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="JWT public key not found",
        )


def require_role(required_role: str):
    """Dependency factory to enforce role-based access control.
    
    Usage:
        @router.get("/me")
        async def get_profile(current_user: dict = Depends(require_role("patient"))):
            ...
    """
    async def role_checker(
        current_user: dict = Depends(get_current_user)
    ) -> dict:
        user_roles = current_user.get("roles", [])
        if required_role not in user_roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Access denied. Required role: {required_role}",
            )
        return current_user
    
    return role_checker


async def verify_internal_api_key(
    request: Request,
    x_internal_api_key: Optional[str] = None
) -> bool:
    """Verify service-to-service API key for internal endpoints.
    
    Internal endpoints should NOT be exposed through the API Gateway.
    They are only accessible directly between microservices.
    """
    settings = get_settings()
    
    # Get from header if not passed directly
    if x_internal_api_key is None:
        x_internal_api_key = request.headers.get("X-Internal-API-Key")
    
    if not x_internal_api_key or x_internal_api_key != settings.INTERNAL_API_KEY:
        raise InternalAPIKeyError()
    
    return True

"""Audit logging service for HIPAA compliance."""
from typing import Optional
from uuid import UUID

from sqlalchemy.ext.asyncio import AsyncSession

from app.models.audit import ProfileAuditLog


class AuditService:
    """Service for logging all access to protected health information."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def log_action(
        self,
        user_id: UUID,
        action: str,
        table_name: str,
        record_id: Optional[UUID] = None,
        old_values: Optional[dict] = None,
        new_values: Optional[dict] = None,
        performed_by: Optional[UUID] = None,
        ip_address: Optional[str] = None,
        user_agent: Optional[str] = None,
    ) -> ProfileAuditLog:
        """Log an action on protected health information.
        
        Args:
            user_id: The patient/doctor whose data was accessed
            action: One of: create, update, delete, view
            table_name: The database table affected
            record_id: The specific record ID
            old_values: Previous values (for updates/deletes)
            new_values: New values (for creates/updates)
            performed_by: User who performed the action (defaults to user_id)
            ip_address: Client IP address
            user_agent: Client user agent string
            
        Returns:
            The created audit log entry
        """
        log = ProfileAuditLog(
            user_id=user_id,
            action=action,
            table_name=table_name,
            record_id=record_id,
            old_values=old_values,
            new_values=new_values,
            performed_by=performed_by or user_id,
            ip_address=ip_address,
            user_agent=user_agent,
        )
        self.db.add(log)
        await self.db.flush()
        return log
    
    async def log_view(
        self,
        user_id: UUID,
        table_name: str,
        performed_by: UUID,
        ip_address: Optional[str] = None,
        user_agent: Optional[str] = None,
    ) -> ProfileAuditLog:
        """Convenience method for logging data views."""
        return await self.log_action(
            user_id=user_id,
            action="view",
            table_name=table_name,
            performed_by=performed_by,
            ip_address=ip_address,
            user_agent=user_agent,
        )
    
    async def log_create(
        self,
        user_id: UUID,
        table_name: str,
        record_id: UUID,
        new_values: dict,
        performed_by: UUID,
        ip_address: Optional[str] = None,
        user_agent: Optional[str] = None,
    ) -> ProfileAuditLog:
        """Convenience method for logging creates."""
        return await self.log_action(
            user_id=user_id,
            action="create",
            table_name=table_name,
            record_id=record_id,
            new_values=new_values,
            performed_by=performed_by,
            ip_address=ip_address,

        )

    # =====================================================================
    # PATIENT PROFILE OPERATIONS
    # =====================================================================
    
    async def get_patient_profile(self, user_id: UUID) -> Optional[PatientProfile]:
        """Get patient profile with all related data eagerly loaded."""
        result = await self.db.execute(
            select(PatientProfile)
            .options(
                selectinload(PatientProfile.allergies),
                selectinload(PatientProfile.medications),
                selectinload(PatientProfile.conditions),
            )
            .where(PatientProfile.user_id == user_id)
        )
        return result.scalar_one_or_none()
    
    async def patient_profile_exists(self, user_id: UUID) -> bool:
        """Check if a patient profile exists."""
        result = await self.db.execute(
            select(exists().where(PatientProfile.user_id == user_id))
        )
        return result.scalar()
    
    async def create_patient_profile(
        self, user_id: UUID, data: PatientProfileCreate
    ) -> PatientProfile:
        """Create a new patient profile."""
        # Check if profile already exists
        if await self.patient_profile_exists(user_id):
            raise ProfileAlreadyExistsError()
        
        profile = PatientProfile(
            user_id=user_id,
            **data.model_dump(),
        )
        self.db.add(profile)
        await self.db.flush()
        return profile
    
    async def update_patient_profile(
        self, user_id: UUID, data: PatientProfileUpdate
    ) -> Optional[PatientProfile]:
        """Update patient profile fields. Only updates provided fields."""
        result = await self.db.execute(
            select(PatientProfile).where(PatientProfile.user_id == user_id)
        )
        profile = result.scalar_one_or_none()
        if not profile:
            return None
        
        update_data = data.model_dump(exclude_unset=True)
        for field, value in update_data.items():
            setattr(profile, field, value)
        
        await self.db.flush()
        return profile

    async def get_medical_summary(self, user_id: UUID) -> Optional[MedicalSummaryResponse]:
        """Generate condensed medical summary for AI services.
        
        This is called automatically at the start of each consultation
        so the patient does not have to re-enter known information.
        """
        profile = await self.get_patient_profile(user_id)
        if not profile:
            return None
        
        # Calculate age from birth_date
        today = date.today()
        age = today.year - profile.birth_date.year
        if (today.month, today.day) < (profile.birth_date.month, profile.birth_date.day):
            age -= 1
        
        # Calculate BMI
        bmi = None
        if profile.height_cm and profile.weight_kg:
            height_m = profile.height_cm / 100
            bmi = round(profile.weight_kg / (height_m ** 2), 1)
        
        # Filter active records only
        active_medications = [m for m in profile.medications if m.is_active]
        active_conditions = [c for c in profile.conditions if c.is_active]
        
        return MedicalSummaryResponse(
            user_id=profile.user_id,
            full_name=profile.full_name,
            age=age,
            gender=profile.gender or "unknown",
            bmi=bmi,
            blood_type=profile.blood_type,
            allergies=[
                {"name": a.allergy_name, "severity": a.severity}
                for a in profile.allergies
            ],
            current_medications=[
                {
                    "name": m.medication_name,
                    "dosage": m.dosage,
                    "frequency": m.frequency,
                    "prescribed_by": m.prescribed_by,
                }
                for m in active_medications
            ],
            chronic_conditions=[
                {
                    "name": c.condition_name,
                    "severity": c.severity,
                    "icd10_code": c.icd10_code,
                }
                for c in active_conditions
            ],
        )

"""Object storage service for file uploads (MinIO/S3-compatible)."""
import uuid
from typing import Optional

import boto3
from botocore.client import Config
from botocore.exceptions import ClientError
from fastapi import UploadFile

from app.config import get_settings
from app.exceptions import InvalidFileError, FileTooLargeError


class StorageService:
    """Service for uploading and managing files in object storage."""
    
    def __init__(self):
        settings = get_settings()
        self.settings = settings
        
        protocol = "https" if settings.MINIO_SECURE else "http"
        endpoint_url = f"{protocol}://{settings.MINIO_ENDPOINT}"
        
        self.client = boto3.client(
            "s3",
            endpoint_url=endpoint_url,
            aws_access_key_id=settings.MINIO_ACCESS_KEY,
            aws_secret_access_key=settings.MINIO_SECRET_KEY,
            config=Config(signature_version="s3v4"),
            region_name="us-east-1",
        )
        self.bucket = settings.MINIO_BUCKET_NAME
        self._ensure_bucket_exists()
    
    def _ensure_bucket_exists(self):
        """Create bucket if it does not exist."""
        try:
            self.client.head_bucket(Bucket=self.bucket)
        except ClientError as e:
            error_code = e.response["Error"]["Code"]
            if error_code == "404":
                self.client.create_bucket(Bucket=self.bucket)
            else:
                raise
    
    def validate_file(self, file: UploadFile) -> None:
        """Validate uploaded file type and size."""
        # Check file type
        if file.content_type not in self.settings.ALLOWED_IMAGE_TYPES:
            raise InvalidFileError(
                f"Invalid file type: {file.content_type}. "
                f"Allowed types: {', '.join(self.settings.ALLOWED_IMAGE_TYPES)}"
            )
    
    async def upload_avatar(
        self, file: UploadFile, user_id: str, user_type: str = "patient"
    ) -> tuple[str, str]:
        """Upload an avatar image to object storage.
        
        Args:
            file: The uploaded file
            user_id: UUID of the user
            user_type: "patient" or "doctor"
            
        Returns:
            Tuple of (public_url, object_key)
        """
        self.validate_file(file)
        
        # Generate unique object key
        ext = file.filename.split(".")[-1] if "." in file.filename else "jpg"
        object_key = f"avatars/{user_type}s/{user_id}/{uuid.uuid4()}.{ext}"
        
        # Read file content
        content = await file.read()
        
        # Check file size
        max_size = self.settings.MAX_AVATAR_SIZE_MB * 1024 * 1024
        if len(content) > max_size:
            raise FileTooLargeError(self.settings.MAX_AVATAR_SIZE_MB)
        
        # Upload to storage
        self.client.put_object(
            Bucket=self.bucket,
            Key=object_key,
            Body=content,
            ContentType=file.content_type,
            Metadata={
                "uploaded-by": "profile-service",
                "user-id": user_id,
                "user-type": user_type,
            },
        )
        
        # Generate presigned URL (valid for 7 days)
        url = self.client.generate_presigned_url(
            "get_object",
            Params={"Bucket": self.bucket, "Key": object_key},
            ExpiresIn=3600 * 24 * 7,
        )
        
        return url, object_key
    
    async def delete_file(self, object_key: str) -> bool:
        """Delete a file from object storage.
        
        Returns:
            True if deleted successfully, False if not found.
        """
        try:
            self.client.delete_object(Bucket=self.bucket, Key=object_key)
            return True
        except ClientError:
            return False
    
    async def get_file_url(self, object_key: str, expires_in: int = 3600) -> Optional[str]:
        """Generate a presigned URL for accessing a file.
        
        Args:
            object_key: The storage object key
            expires_in: URL expiration time in seconds
            
        Returns:
            Presigned URL or None if object not found
        """
        try:
            url = self.client.generate_presigned_url(
                "get_object",
                Params={"Bucket": self.bucket, "Key": object_key},
                ExpiresIn=expires_in,
            )
            return url
        except ClientError:
            return None

"""Utility helper functions."""
from datetime import date
from uuid import UUID


def calculate_age(birth_date: date) -> int:
    """Calculate age from birth date."""
    today = date.today()
    age = today.year - birth_date.year
    if (today.month, today.day) < (birth_date.month, birth_date.day):
        age -= 1
    return age


def calculate_bmi(height_cm: int, weight_kg: int) -> float:
    """Calculate BMI from height (cm) and weight (kg)."""
    height_m = height_cm / 100
    return round(weight_kg / (height_m ** 2), 1)


def is_valid_uuid(value: str) -> bool:
    """Check if a string is a valid UUID."""
    try:
        UUID(value)
        return True
    except ValueError:
        return False
