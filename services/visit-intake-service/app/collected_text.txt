"""Application configuration using Pydantic Settings.

Backward-compatible: keeps BOTH the old AI_NLP_* names (so the original
evidence_extraction.py and nlp_client.py keep working) AND the new
AI_TRIAGE_* names (so the updated nlp_client.py that targets
ai-triage-service also works).

NEW: AI_DIAGNOSIS_* settings for the diagnosis flow (doctor claims visit →
visit-intake-service calls ai-triage-service POST /visits/{id}/diagnose,
stores the predictions, exposes them via GET /visits/{id}/ai-diagnosis).
"""
from functools import lru_cache
from typing import Optional

from pydantic_settings import BaseSettings


class Settings(BaseSettings):
    """Application settings loaded from environment variables."""

    APP_NAME: str = "Visit Intake Service"
    DEBUG: bool = False

    # Database (database-per-service: visits_db)
    DATABASE_URL: str = "postgresql+asyncpg://postgres:password@localhost:5432/visits_db"

    # JWT
    JWT_PUBLIC_KEY_PATH: str = "./keys/public.pem"
    JWT_ALGORITHM: str = "RS256"

    # Object Storage (MinIO/S3-compatible)
    MINIO_ENDPOINT: str = "minio:9000"
    MINIO_ACCESS_KEY: str = ""
    MINIO_SECRET_KEY: str = ""
    MINIO_BUCKET_NAME: str = "visit-files"
    MINIO_SECURE: bool = False

    # Internal API Security
    INTERNAL_API_KEY: str = ""

    # medical-record-service (sync hook)
    MEDICAL_RECORD_SERVICE_URL: str = "http://localhost:8000"

    # ── AI orchestration ──────────────────────────────────────────────
    # NEW names (preferred — used by the updated nlp_client.py)
    AI_TRIAGE_SERVICE_URL: str = "http://ai-triage-service:8000"
    AI_TRIAGE_ENABLED: bool = True
    AI_TRIAGE_TIMEOUT_SECONDS: float = 30.0

    # OLD names (kept for backward compat with the original evidence_extraction.py
    # and any other code that hasn't been updated yet).
    AI_NLP_SERVICE_URL: str = "http://ai-triage-service:8000"  # mirror triage URL for compat
    AI_NLP_ENABLED: bool = True
    AI_NLP_TIMEOUT_SECONDS: float = 30.0

    # Shared by both old and new code paths
    AI_NLP_TEXT_MAX_CHARS: int = 5000
    AI_NLP_THRESHOLD: Optional[float] = None   # None -> use the service default (0.3)

    # ── NEW — Diagnosis flow ──────────────────────────────────────────
    # When a doctor claims a visit, visit-intake-service calls
    # ai-triage-service POST /visits/{visit_id}/diagnose to get disease
    # predictions, stores them in visits_db.ai_diagnosis_suggestions, and
    # exposes them via GET /visits/{visit_id}/ai-diagnosis.
    AI_DIAGNOSIS_ENABLED: bool = True
    AI_DIAGNOSIS_TOP_K: int = 10
    AI_DIAGNOSIS_TIMEOUT_SECONDS: float = 45.0   # diagnosis takes longer than extract

    # If true, the doctor's claim-visit endpoint automatically triggers AI
    # diagnosis. If false, the doctor must hit POST /visits/{id}/diagnose
    # explicitly (useful for development / A/B testing).
    AI_DIAGNOSIS_TRIGGER_ON_CLAIM: bool = True

    # File Upload
    MAX_ATTACHMENT_SIZE_MB: int = 25
    ALLOWED_ATTACHMENT_TYPES: list[str] = [
        "image/jpeg",
        "image/png",
        "image/webp",
        "application/pdf",
        "audio/mpeg",
        "audio/wav",
        "video/mp4",
    ]

    class Config:
        env_file = ".env"
        env_file_encoding = "utf-8"
        extra = "ignore"   # ← tolerate unknown env vars instead of crashing


@lru_cache()
def get_settings() -> Settings:
    """Return cached settings instance."""
    return Settings()

    autoflush=False,

class Base(DeclarativeBase):

async def get_db():
    """Yield an async database session for FastAPI dependency injection."""
    async with AsyncSessionLocal() as session:
        try:
            yield session
            await session.commit()
        except Exception:
            await session.rollback()
            raise
        finally:
            await session.close()

"""Authentication dependencies for FastAPI."""
from typing import Optional

from fastapi import Depends, HTTPException, status, Request
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import JWTError, jwt

from app.config import get_settings
from app.exceptions import InternalAPIKeyError

security = HTTPBearer(auto_error=False)


async def get_current_user(
    request: Request,
    credentials: Optional[HTTPAuthorizationCredentials] = Depends(security)
) -> dict:
    """Validate user identity from JWT OR trusted Gateway headers.

    Priority:
    1. If x-gateway-request header exists -> Trust x-user-id/x-user-role (Gateway already validated)
    2. Otherwise -> Validate JWT locally (direct access or other services)
    """
    settings = get_settings()

    is_gateway_request = request.headers.get("x-gateway-request") == "true"
    user_id_from_gateway = request.headers.get("x-user-id")
    user_role_from_gateway = request.headers.get("x-user-role")

    if is_gateway_request and user_id_from_gateway:
        return {
            "sub": user_id_from_gateway,
            "role": user_role_from_gateway,
            "source": "gateway",
        }

    if not credentials:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authorization header missing",
            headers={"WWW-Authenticate": "Bearer"},
        )

    try:
        with open(settings.JWT_PUBLIC_KEY_PATH, "r") as f:
            public_key = f.read()

        payload = jwt.decode(
            credentials.credentials,
            public_key,
            algorithms=[settings.JWT_ALGORITHM],
        )

        user_id: str = payload.get("sub")
        if user_id is None:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid token: missing subject",
            )

        payload["source"] = "jwt"
        return payload

    except JWTError as e:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=f"Invalid token: {str(e)}",
            headers={"WWW-Authenticate": "Bearer"},
        )
    except FileNotFoundError:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="JWT public key not found",
        )


def require_role(required_role: str):
    """Dependency factory to enforce role-based access control."""
    async def role_checker(
        current_user: dict = Depends(get_current_user)
    ) -> dict:
        user_roles = current_user.get("roles", [])
        if isinstance(user_roles, str):
            user_roles = [user_roles]
        user_role = current_user.get("role")
        if user_role:
            user_roles.append(user_role)

        if required_role not in user_roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Access denied. Required role: {required_role}",
            )
        return current_user

    return role_checker


async def verify_internal_api_key(
    request: Request,
    x_internal_api_key: Optional[str] = None
) -> bool:
    """Verify service-to-service API key for internal endpoints."""
    settings = get_settings()

    if x_internal_api_key is None:
        x_internal_api_key = request.headers.get("X-Internal-API-Key")

    if not x_internal_api_key or x_internal_api_key != settings.INTERNAL_API_KEY:
        raise InternalAPIKeyError()

    return True

async def get_participant_visit(
    visit_id: UUID,
    current_user: dict,
    db: AsyncSession,
):
    """Allow only the owning patient or the assigned doctor."""
    service = VisitService(db)
    visit = await service.get_visit(visit_id)
    if not visit:
        raise VisitNotFoundError()

    user_id = UUID(current_user["sub"])
    roles = current_user.get("roles", [])
    if isinstance(roles, str):
        roles = [roles]
    role = current_user.get("role")
    if role:
        roles.append(role)

    is_owner = "patient" in roles and visit.patient_id == user_id
    is_doctor = "doctor" in roles and visit.doctor_id == user_id
    if not (is_owner or is_doctor):
        raise ForbiddenAccessError("You are not a participant of this visit")
    return visit, "patient" if is_owner else "doctor"

async def _trigger_and_store_diagnosis(
    visit_id: UUID,
    db: AsyncSession,
    patient_sex: str | None = None,
    patient_age: int | None = None,
    top_k: int | None = None,
    reextract: bool = False,
) -> AIDiagnosisResponse:
    """Call ai-triage-service, wipe old suggestions, store new ones.

    Used by:
      - POST /visits/{id}/diagnose          (this router)
      - The doctor's claim-visit endpoint   (imported from app.routers.doctor)

    Raises DiagnosisServiceUnavailableError if ai-triage-service is down.
    """
    if not settings.AI_DIAGNOSIS_ENABLED:
        raise DiagnosisServiceUnavailableError(
            "AI diagnosis is disabled (AI_DIAGNOSIS_ENABLED=false)"
        )

    result = await trigger_visit_diagnosis(
        visit_id=visit_id,
        patient_sex=patient_sex,
        patient_age=patient_age,
        top_k=top_k,
        reextract=reextract,
    )
    if result is None:
        raise DiagnosisServiceUnavailableError(
            "ai-triage-service is unreachable or returned an error"
        )

    # Wipe old suggestions for this visit (we always store the latest set)
    await db.execute(
        delete(AIDiagnosisSuggestion).where(
            AIDiagnosisSuggestion.visit_id == visit_id
        )
    )

    # Insert new ones (rank = 1-based position in predictions[])
    rows = []
    for idx, p in enumerate(result.predictions, start=1):
        row = AIDiagnosisSuggestion(
            visit_id=visit_id,
            disease_id=p.disease_id,
            disease_name_fa=p.disease_name_fa,
            disease_name_en=p.disease_name_en,
            icd10=p.icd10,
            ml_probability=p.ml_probability,
            rule_score=p.rule_score,
            combined_score=p.combined_score,
            is_emergency=p.is_emergency,
            severity=p.severity,
            rank=idx,
            matched_evidences=[m.model_dump() for m in p.matched_evidences],
            absent_evidences=[a.model_dump() for a in p.absent_evidences],
        )
        db.add(row)
        rows.append(row)

    await db.commit()
    await db.refresh(row for row in rows)  # ensures id/created_at are populated

    logger.info(
        "Stored %d AI diagnosis suggestions for visit %s (source=%s)",
        len(rows),
        visit_id,
        result.evidence_source,
    )

    return AIDiagnosisResponse(
        visit_id=visit_id,
        suggestions=[
            AIDiagnosisSuggestionSchema(
                id=str(row.id),
                disease_id=row.disease_id,
                disease_name_fa=row.disease_name_fa,
                disease_name_en=row.disease_name_en,
                icd10=row.icd10,
                ml_probability=row.ml_probability,
                rule_score=row.rule_score,
                combined_score=row.combined_score,
                is_emergency=row.is_emergency,
                severity=row.severity,
                matched_evidences=row.matched_evidences,
                absent_evidences=row.absent_evidences,
                rank=row.rank,
            )
            for row in rows
        ],
        evidence_source=result.evidence_source,
        patient_sex=result.patient_sex,
        model_version=result.model_version,
        rule_engine_version=result.rule_engine_version,
        diagnosed_at=datetime.now(timezone.utc),
        input_evidence_count=result.input_evidence_count,
    )

@router.post(
    "/visits/{visit_id}/final-diagnosis",
    response_model=FinalDiagnosisResponse,
    status_code=status.HTTP_201_CREATED,
)
async def submit_final_diagnosis(
    visit_id: UUID,
    data: FinalDiagnosisRequest,
    response: Response,
    db: AsyncSession = Depends(get_db),
    current_user: dict = Depends(require_role("doctor")),
):
    """Submit the doctor's final diagnosis for a visit.

    The doctor can either:
      - pick one of the AI suggestions (selected_suggestion_id), or
      - enter a manual diagnosis (manual_disease_name + manual_icd10).

    At least one source must be provided. The visit's status is moved to
    "completed" atomically with the diagnosis submission.

    Auth: JWT with role=doctor. Only the doctor assigned to this visit
    can submit (the doctor router enforces assignment before calling us).
    """
    # Validate that the doctor is assigned to this visit
    # (We import lazily to avoid a circular import with app.routers.doctor.)
    from app.routers.doctor import _assert_doctor_owns_visit  # type: ignore
    await _assert_doctor_owns_visit(visit_id, current_user, db)

    # Validate the request
    if not data.selected_suggestion_id and not data.manual_disease_name:
        raise DiagnosisError(
            "Either selected_suggestion_id or manual_disease_name must be provided"
        )

    # Resolve the chosen disease details
    selected_suggestion: AIDiagnosisSuggestion | None = None
    if data.selected_suggestion_id:
        selected_suggestion = (
            await db.execute(
                select(AIDiagnosisSuggestion).where(
                    AIDiagnosisSuggestion.id == UUID(data.selected_suggestion_id),
                    AIDiagnosisSuggestion.visit_id == visit_id,
                )
            )
        ).scalars().first()
        if not selected_suggestion:
            raise NotFoundError(
                f"AI suggestion {data.selected_suggestion_id} not found for visit {visit_id}"
            )

    # If both are provided, the manual entry wins (doctor override)
    if data.manual_disease_name:
        disease_name_fa = data.manual_disease_name
        disease_name_en = None
        icd10 = data.manual_icd10
        disease_id = None
        selected_suggestion_id = None
    else:
        assert selected_suggestion is not None
        disease_name_fa = selected_suggestion.disease_name_fa
        disease_name_en = selected_suggestion.disease_name_en
        icd10 = selected_suggestion.icd10
        disease_id = selected_suggestion.disease_id
        selected_suggestion_id = selected_suggestion.id

    # Idempotency: if there's already a final diagnosis for this visit,
    # update it instead of erroring out (so the doctor can revise).
    existing = (
        await db.execute(
            select(FinalDiagnosis).where(FinalDiagnosis.visit_id == visit_id)
        )
    ).scalars().first()

    is_update = existing is not None
    if existing:
        existing.disease_name_fa = disease_name_fa
        existing.disease_name_en = disease_name_en
        existing.icd10 = icd10
        existing.disease_id = disease_id
        existing.selected_suggestion_id = selected_suggestion_id
        existing.notes = data.notes
        existing.requires_follow_up = data.requires_follow_up
        existing.follow_up_days = data.follow_up_days
        existing.doctor_id = UUID(str(current_user["sub"]))
        existing.created_at = datetime.now(timezone.utc)
        row = existing
    else:
        row = FinalDiagnosis(
            visit_id=visit_id,
            doctor_id=UUID(str(current_user["sub"])),
            disease_name_fa=disease_name_fa,
            disease_name_en=disease_name_en,
            icd10=icd10,
            disease_id=disease_id,
            selected_suggestion_id=selected_suggestion_id,
            notes=data.notes,
            requires_follow_up=data.requires_follow_up,
            follow_up_days=data.follow_up_days,
        )
        db.add(row)

    # Move the visit status to "completed". If your project has a Visit ORM
    # model, replace this raw SQL with a proper ORM update.
    await db.execute(
        text("UPDATE visits SET status = 'completed', completed_at = NOW() WHERE id = :vid"),
        {"vid": visit_id},
    )

    await db.commit()
    await db.refresh(row)

    logger.info(
        "Final diagnosis submitted for visit %s by doctor %s: disease=%s icd10=%s",
        visit_id,
        current_user.get("sub"),
        disease_name_fa,
        icd10,
    )

    # Override the response status code: 201 for new, 200 for update.
    if is_update:
        response.status_code = status.HTTP_200_OK

    return FinalDiagnosisResponse(
        id=str(row.id),
        visit_id=row.visit_id,
        disease_name_fa=row.disease_name_fa,
        disease_name_en=row.disease_name_en,
        icd10=row.icd10,
        disease_id=row.disease_id,
        selected_suggestion_id=str(row.selected_suggestion_id) if row.selected_suggestion_id else None,
        notes=row.notes,
        requires_follow_up=row.requires_follow_up,
        follow_up_days=row.follow_up_days,
        doctor_id=str(row.doctor_id),
        created_at=row.created_at,
        visit_status="completed",
    )

async def trigger_visit_diagnosis(

"""Glue between visit intake and ai-nlp-service.

collect_visit_text()       -> aggregates the patient's Persian intake text
run_extraction_for_visit() -> text -> POST /extract -> AI symptom rows

The results are stored through the existing bulk_add_ai_symptoms() flow, so
the whole platform keeps consuming VisitSymptom rows exactly as before.
"""
import json
import logging
from typing import List, Optional
from uuid import UUID

from sqlalchemy.ext.asyncio import AsyncSession

from app.config import get_settings
from app.exceptions import NLPServiceUnavailableError
from app.schemas.clinical import AISymptomInput, AISymptomsBulkInput
from app.schemas.nlp import DetectedEvidence, EvidenceExtractionResponse
from app.services.nlp_client import extract_evidences
from app.services.visit_service import VisitService

logger = logging.getLogger(__name__)

MAX_SYMPTOM_NAME = 150     # VisitSymptom.symptom_name column width
NOTES_MAX_CHARS = 4000     # hard cap for the traceability JSON stored in notes


def collect_visit_text(messages) -> str:
    """Aggregate the patient's Persian intake text in chronological order.

    Text messages contribute their content; file messages contribute only
    their caption (attachment bytes are not OCR'd here). Non-patient messages
    (ai / system follow-ups) are excluded — the model must only see the
    patient's own words.
    """
    parts = [
        (m.content or "").strip()
        for m in sorted(messages, key=lambda m: m.sequence)
        if m.sender_type == "patient" and (m.content or "").strip()
    ]
    return "\n".join(parts)


def _evidence_to_symptom(evidence: DetectedEvidence) -> AISymptomInput:
    """Map one detected evidence to an AI symptom row.

    symptom_name prefers the Persian display name (falls back to the English
    name, then the raw evidence_id) so the doctor UI stays readable; the
    canonical evidence_id plus the full metadata travel in `notes` as JSON.
    """
    display_name = (evidence.name_fa or evidence.name or evidence.evidence_id)
    display_name = display_name.strip()[:MAX_SYMPTOM_NAME]

    trace = {"evidence_id": evidence.evidence_id}
    if evidence.code_question:
        trace["code_question"] = evidence.code_question
    if evidence.question_fa:
        trace["question_fa"] = evidence.question_fa
    if evidence.question_en:
        trace["question_en"] = evidence.question_en
    if evidence.data_type:
        trace["data_type"] = evidence.data_type
    notes = json.dumps(trace, ensure_ascii=False, sort_keys=True)[:NOTES_MAX_CHARS]

    return AISymptomInput(
        symptom_name=display_name,
        confidence=max(0.0, min(1.0, evidence.confidence)),
        notes=notes,
    )


async def run_extraction_for_visit(
    visit_id: UUID,
    db: AsyncSession,
    *,
    fail_safe: bool = True,
) -> Optional[EvidenceExtractionResponse]:
    """Run evidence extraction for a visit and store the results as AI symptoms.

    fail_safe=True  -> any failure is logged and None is returned. Used inside
                       POST /visits/{id}/submit: an unavailable NLP service
                       must never block the submission.
    fail_safe=False -> raises NLPServiceUnavailableError on failure. Used by
                       POST /visits/{id}/extract-evidences, where the caller
                       explicitly asked for the extraction and wants feedback.
    """
    settings = get_settings()

    if not settings.AI_NLP_ENABLED:
        logger.info("AI NLP integration disabled — skipping extraction for visit %s", visit_id)
        if fail_safe:
            return None
        raise NLPServiceUnavailableError(
            "AI NLP integration is disabled (AI_NLP_ENABLED=false)"
        )

    service = VisitService(db)
    messages = await service.list_messages(visit_id)
    text = collect_visit_text(messages)

    if not text:
        logger.info("Visit %s has no patient text to analyse", visit_id)
        return EvidenceExtractionResponse(
            visit_id=visit_id,
            text_chars=0,
            detected=[],
            stored_count=0,
            duplicate_count=0,
        )

    if len(text) > settings.AI_NLP_TEXT_MAX_CHARS:
        logger.info(
            "Visit %s text truncated from %d to %d chars before sending",
            visit_id, len(text), settings.AI_NLP_TEXT_MAX_CHARS,
        )
        text = text[: settings.AI_NLP_TEXT_MAX_CHARS]

    result = await extract_evidences(text)
    if result is None:
        if fail_safe:
            logger.warning(
                "Evidence extraction failed for visit %s — continuing without AI symptoms",
                visit_id,
            )
            return None
        raise NLPServiceUnavailableError()

    symptoms = [_evidence_to_symptom(e) for e in result.detected]
    stored = await service.bulk_add_ai_symptoms(
        visit_id,
        AISymptomsBulkInput(
            symptoms=symptoms,
            model_version=(result.model_version or "")[:50] or None,
        ),
    )

    outcome = EvidenceExtractionResponse(
        visit_id=visit_id,
        text_chars=len(text),
        detected=result.detected,
        stored_count=len(stored),
        duplicate_count=len(result.detected) - len(stored),
        model_version=result.model_version,
        inference_time_ms=result.inference_time_ms,
    )
    logger.info(
        "Extraction for visit %s: %d detected, %d stored, %d duplicate(s)",
        visit_id, len(result.detected), len(stored), outcome.duplicate_count,
    )
    return outcome

async def extract_evidences(text: str) -> Optional[ExtractResult]:
    """Send Persian free text to ai-triage-service and return the parsed result.

    Returns None when:
      - AI_TRIAGE_ENABLED is false,
      - AI_TRIAGE_SERVICE_URL is empty,
      - the call fails (network error / timeout / non-200 / malformed payload).
    """
    settings = get_settings()

    if not settings.AI_TRIAGE_ENABLED:
        logger.info("AI_TRIAGE_ENABLED is false — skipping evidence extraction")
        return None

    if not settings.AI_TRIAGE_SERVICE_URL:
        logger.warning("AI_TRIAGE_SERVICE_URL is not set — skipping evidence extraction")
        return None

    payload = {"text": text}
    if settings.AI_NLP_THRESHOLD is not None:
        payload["threshold"] = settings.AI_NLP_THRESHOLD

    headers = {}
    if settings.INTERNAL_API_KEY:
        headers["X-Internal-API-Key"] = settings.INTERNAL_API_KEY

    # ai-triage-service exposes POST /extract-evidences (internal, X-Internal-API-Key)
    url = f"{settings.AI_TRIAGE_SERVICE_URL.rstrip('/')}/extract-evidences"
    try:
        async with httpx.AsyncClient(timeout=settings.AI_TRIAGE_TIMEOUT_SECONDS) as client:
            resp = await client.post(url, json=payload, headers=headers)
    except httpx.HTTPError as exc:
        logger.warning("ai-triage-service call failed (%s): %s", url, exc)
        return None

    if resp.status_code != 200:
        logger.warning(
            "ai-triage-service returned HTTP %s: %s",
            resp.status_code,
            resp.text[:500],
        )
        return None

    try:
        body = resp.json()
    except ValueError:
        logger.warning("ai-triage-service returned a non-JSON body")

    # use it (the frontend chat reads it via the preview endpoint).
    try:

            DetectedEvidence.model_validate(d) for d in body.get("detected", [])

        detected=detected,
        threshold=body.get("threshold"),
        truncated=body.get("truncated"),
        model_version=body.get("model_version"),
        inference_time_ms=body.get("inference_time_ms"),
    )

    logger.info(
        "ai-triage-service: %d evidence(s) in %s ms (priority=%s, model_version=%s)",
        len(result.detected),
        result.inference_time_ms,
        body.get("priority"),
        result.model_version,
    )
    return result

"""Object storage service for visit file attachments (MinIO/S3-compatible)."""
import uuid

import boto3
from botocore.client import Config
from botocore.exceptions import ClientError
from fastapi import UploadFile

from app.config import get_settings
from app.exceptions import InvalidFileError, FileTooLargeError


class StorageService:
    """Service for uploading/managing files exchanged inside visit messages."""

    def __init__(self):
        settings = get_settings()
        self.settings = settings

        protocol = "https" if settings.MINIO_SECURE else "http"
        endpoint_url = f"{protocol}://{settings.MINIO_ENDPOINT}"

        self.client = boto3.client(
            "s3",
            endpoint_url=endpoint_url,
            aws_access_key_id=settings.MINIO_ACCESS_KEY,
            aws_secret_access_key=settings.MINIO_SECRET_KEY,
            config=Config(signature_version="s3v4"),
            region_name="us-east-1",
        )
        self.bucket = settings.MINIO_BUCKET_NAME
        self._ensure_bucket_exists()

    def _ensure_bucket_exists(self):
        try:
            self.client.head_bucket(Bucket=self.bucket)
        except ClientError as e:
            if e.response["Error"]["Code"] == "404":
                self.client.create_bucket(Bucket=self.bucket)
            else:
                raise

    def validate_file(self, file: UploadFile) -> None:
        if file.content_type not in self.settings.ALLOWED_ATTACHMENT_TYPES:
            raise InvalidFileError(
                f"Invalid file type: {file.content_type}. "
                f"Allowed types: {', '.join(self.settings.ALLOWED_ATTACHMENT_TYPES)}"
            )

    async def upload_attachment(self, file: UploadFile, visit_id: str) -> dict:
        """Upload a file attachment for a visit message.

        Returns a dict with attachment metadata ready to pass to VisitService.add_message.
        """
        self.validate_file(file)
        content = await file.read()

        max_size = self.settings.MAX_ATTACHMENT_SIZE_MB * 1024 * 1024
        if len(content) > max_size:
            raise FileTooLargeError(self.settings.MAX_ATTACHMENT_SIZE_MB)

        ext = file.filename.split(".")[-1] if "." in file.filename else "bin"
        object_key = f"visits/{visit_id}/attachments/{uuid.uuid4()}.{ext}"

        self.client.put_object(
            Bucket=self.bucket,
            Key=object_key,
            Body=content,
            ContentType=file.content_type,
            Metadata={"uploaded-by": "visit-intake-service", "visit-id": visit_id},
        )

        url = self.client.generate_presigned_url(
            "get_object",
            Params={"Bucket": self.bucket, "Key": object_key},
            ExpiresIn=3600 * 24 * 7,
        )

        return {
            "attachment_name": file.filename,
            "attachment_content_type": file.content_type,
            "attachment_size": len(content),
            "attachment_key": object_key,
            "attachment_url": url,
        }

    def get_file_url(self, object_key: str, expires_in: int = 3600 * 24 * 7) -> str | None:
        """Fresh presigned URL (stored URLs expire, so regenerate on read)."""
        try:
            return self.client.generate_presigned_url(
                "get_object",
                Params={"Bucket": self.bucket, "Key": object_key},
                ExpiresIn=expires_in,
            )
        except ClientError:
            return None

    async def delete_file(self, object_key: str) -> bool:
        try:
            self.client.delete_object(Bucket=self.bucket, Key=object_key)
            return True
        except ClientError:
            return False

"""Business logic for visits, messages, symptoms, AI results and prescriptions."""
from datetime import datetime
from typing import Optional
from uuid import UUID

from sqlalchemy import delete, func, select
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy.orm import selectinload

from app.models.visit import Visit
from app.models.message import VisitMessage
from app.models.clinical import (
    VisitSymptom,
    AIDiseaseSuggestion,
    AIPrescriptionDraft,
    FinalPrescription,
    PrescriptionItem,
)
from app.models.chat import ChatMessage
from app.schemas.visit import VisitCreate
from app.schemas.clinical import (
    SymptomCreate,
    AISymptomsBulkInput,
    DiseaseSuggestionsBulkInput,
    AIPrescriptionDraftInput,
    FinalPrescriptionCreate,
)
from app.exceptions import InvalidVisitStateError


# Valid status transitions (visit lifecycle)
ALLOWED_TRANSITIONS = {
    "intake": {"ai_processing", "cancelled"},
    "ai_processing": {"doctor_assigned", "cancelled"},
    "doctor_assigned": {"in_chat", "cancelled"},
    "in_chat": {"prescription_pending", "completed", "cancelled"},
    "prescription_pending": {"completed", "cancelled"},
    "completed": {"closed"},
    "closed": set(),
    "cancelled": set(),
}


class VisitService:
    """Service layer for visit intake operations."""

    def __init__(self, db: AsyncSession):
        self.db = db

    # =====================================================================
    # VISIT LIFECYCLE
    # =====================================================================

    async def create_visit(self, patient_id: UUID, data: VisitCreate) -> Visit:
        visit = Visit(patient_id=patient_id, **data.model_dump())
        self.db.add(visit)
        await self.db.flush()
        return visit

    async def get_visit(self, visit_id: UUID) -> Optional[Visit]:
        """Get a visit with all related data eagerly loaded."""
        result = await self.db.execute(select(Visit).where(Visit.id == visit_id))
        return result.scalar_one_or_none()

    async def list_patient_visits(self, patient_id: UUID) -> list[Visit]:
        result = await self.db.execute(
            select(Visit)
            .where(Visit.patient_id == patient_id)
            .order_by(Visit.started_at.desc())
        )
        return result.scalars().all()

    async def list_doctor_visits(self, doctor_id: UUID) -> list[Visit]:
        result = await self.db.execute(
            select(Visit)
            .where(
                Visit.doctor_id == doctor_id,
                Visit.status.notin_(["closed", "cancelled"]),
            )
            .order_by(Visit.started_at.desc())
        )
        return result.scalars().all()

    async def list_queue(self) -> list[Visit]:
        """Unassigned visits that are ready for a doctor (AI finished processing)."""
        result = await self.db.execute(
            select(Visit)
            .where(Visit.status == "doctor_assigned", Visit.doctor_id.is_(None))
            .order_by(Visit.started_at.asc())
        )
        return result.scalars().all()

    async def transition(self, visit: Visit, new_status: str) -> Visit:
        allowed = ALLOWED_TRANSITIONS.get(visit.status, set())
        if new_status not in allowed:
            raise InvalidVisitStateError(
                f"Cannot move visit from '{visit.status}' to '{new_status}'"
            )
        visit.status = new_status
        now = datetime.utcnow()
        if new_status == "in_chat" and visit.doctor_joined_at is None:
            visit.doctor_joined_at = now
        if new_status == "completed" and visit.completed_at is None:
            visit.completed_at = now
        await self.db.flush()
        return visit

    async def claim(self, visit: Visit, doctor_id: UUID) -> Visit:
        if visit.doctor_id is not None:
            raise InvalidVisitStateError("Visit is already assigned to a doctor")
        if visit.status != "doctor_assigned":
            raise InvalidVisitStateError(
                f"Visit is not ready for a doctor (status: {visit.status})"
            )
        visit.doctor_id = doctor_id
        await self.transition(visit, "in_chat")
        return visit

    async def link_medical_record(self, visit: Visit, medical_record_id: UUID) -> Visit:
        visit.medical_record_id = medical_record_id
        await self.db.flush()
        return visit

    # =====================================================================
    # INTAKE MESSAGES (text / file)
    # =====================================================================

    async def add_message(
        self,
        visit_id: UUID,
        content: Optional[str] = None,
        sender_type: str = "patient",
        message_type: str = "text",
        attachment: Optional[dict] = None,
    ) -> VisitMessage:
        result = await self.db.execute(
            select(func.count())
            .select_from(VisitMessage)
            .where(VisitMessage.visit_id == visit_id)
        )
        sequence = result.scalar_one() + 1

        message = VisitMessage(
            visit_id=visit_id,
            sender_type=sender_type,
            message_type=message_type,
            content=content,
            sequence=sequence,
            sent_at=datetime.utcnow(),
            **(attachment or {}),
        )
        self.db.add(message)
        await self.db.flush()
        return message

    async def list_messages(self, visit_id: UUID) -> list[VisitMessage]:
        result = await self.db.execute(
            select(VisitMessage)
            .where(VisitMessage.visit_id == visit_id)
            .order_by(VisitMessage.sequence.asc())
        )
        return result.scalars().all()

    # =====================================================================
    # SYMPTOMS (manual + AI)
    # =====================================================================

    async def add_symptom(
        self, visit_id: UUID, data: SymptomCreate, source: str = "manual"
    ) -> VisitSymptom:
        symptom = VisitSymptom(visit_id=visit_id, source=source, **data.model_dump())
        self.db.add(symptom)
        await self.db.flush()
        return symptom

    async def bulk_add_ai_symptoms(
        self, visit_id: UUID, payload: AISymptomsBulkInput
    ) -> list[VisitSymptom]:
        """Add AI-detected symptoms, skipping ones already stored for this visit."""
        result = await self.db.execute(
            select(VisitSymptom.symptom_name).where(
                VisitSymptom.visit_id == visit_id,
                VisitSymptom.source == "ai",
            )
        )
        existing = {name.strip().lower() for name in result.scalars().all()}

        added = []
        for item in payload.symptoms:
            if item.symptom_name.strip().lower() in existing:
                continue
            symptom = VisitSymptom(
                visit_id=visit_id,
                source="ai",
                ai_model_version=payload.model_version,
                **item.model_dump(),
            )
            self.db.add(symptom)
            added.append(symptom)
        await self.db.flush()
        return added

    async def list_symptoms(self, visit_id: UUID) -> list[VisitSymptom]:
        result = await self.db.execute(
            select(VisitSymptom)
            .where(VisitSymptom.visit_id == visit_id)
            .order_by(VisitSymptom.created_at.asc())
        )
        return result.scalars().all()

    async def delete_symptom(self, visit_id: UUID, symptom_id: UUID) -> bool:
        result = await self.db.execute(
            select(VisitSymptom).where(
                VisitSymptom.visit_id == visit_id,
                VisitSymptom.id == symptom_id,
                VisitSymptom.source == "manual",  # AI symptoms are only removed by AI re-run
            )
        )
        symptom = result.scalar_one_or_none()
        if not symptom:
            return False
        await self.db.delete(symptom)
        await self.db.flush()
        return True

    # =====================================================================
    # AI DISEASE SUGGESTIONS
    # =====================================================================

    async def replace_disease_suggestions(
        self, visit_id: UUID, payload: DiseaseSuggestionsBulkInput
    ) -> list[AIDiseaseSuggestion]:
        """AI re-runs replace the previous suggestion list entirely."""
        await self.db.execute(
            delete(AIDiseaseSuggestion).where(AIDiseaseSuggestion.visit_id == visit_id)
        )

        suggestions = []
        for item in payload.suggestions:
            suggestion = AIDiseaseSuggestion(
                visit_id=visit_id,
                model_version=payload.model_version,
                **item.model_dump(),
            )
            self.db.add(suggestion)
            suggestions.append(suggestion)
        await self.db.flush()
        return suggestions

    async def list_disease_suggestions(self, visit_id: UUID) -> list[AIDiseaseSuggestion]:
        result = await self.db.execute(
            select(AIDiseaseSuggestion)
            .where(AIDiseaseSuggestion.visit_id == visit_id)
            .order_by(AIDiseaseSuggestion.rank.asc())
        )
        return result.scalars().all()

    async def decide_suggestion(
        self, visit_id: UUID, suggestion_id: UUID, decision: str, doctor_id: UUID
    ) -> Optional[AIDiseaseSuggestion]:
        result = await self.db.execute(
            select(AIDiseaseSuggestion).where(
                AIDiseaseSuggestion.visit_id == visit_id,
                AIDiseaseSuggestion.id == suggestion_id,
            )
        )
        suggestion = result.scalar_one_or_none()
        if not suggestion:
            return None
        suggestion.doctor_decision = decision
        suggestion.decided_by = doctor_id
        suggestion.decided_at = datetime.utcnow()
        await self.db.flush()
        return suggestion

    # =====================================================================
    # AI PRESCRIPTION DRAFT
    # =====================================================================

    async def save_draft(
        self, visit_id: UUID, payload: AIPrescriptionDraftInput
    ) -> AIPrescriptionDraft:
        """Save a new draft; previous drafts are marked superseded."""
        result = await self.db.execute(
            select(AIPrescriptionDraft).where(
                AIPrescriptionDraft.visit_id == visit_id,
                AIPrescriptionDraft.status == "draft",
            )
        )
        for old in result.scalars().all():
            old.status = "superseded"

        draft = AIPrescriptionDraft(
            visit_id=visit_id,
            items=[item.model_dump() for item in payload.items],
            ai_notes=payload.ai_notes,
            model_version=payload.model_version,
        )
        self.db.add(draft)
        await self.db.flush()
        return draft

    async def get_latest_draft(self, visit_id: UUID) -> Optional[AIPrescriptionDraft]:
        result = await self.db.execute(
            select(AIPrescriptionDraft)
            .where(
                AIPrescriptionDraft.visit_id == visit_id,
                AIPrescriptionDraft.status == "draft",
            )
            .order_by(AIPrescriptionDraft.created_at.desc())
        )
        return result.scalars().first()

    # =====================================================================
    # FINAL PRESCRIPTION (doctor)
    # =====================================================================

    async def issue_final_prescription(
        self, visit_id: UUID, doctor_id: UUID, data: FinalPrescriptionCreate
    ) -> FinalPrescription:
        visit = await self.get_visit(visit_id)
        if visit is None:
            raise InvalidVisitStateError("Visit not found")
        if visit.final_prescription is not None:
            raise InvalidVisitStateError("Final prescription already issued for this visit")

        prescription = FinalPrescription(
            visit_id=visit_id,
            doctor_id=doctor_id,
            diagnosis_text=data.diagnosis_text,
            notes=data.notes,
            issued_at=datetime.utcnow(),
        )
        self.db.add(prescription)
        await self.db.flush()

        for index, item in enumerate(data.items):
            self.db.add(
                PrescriptionItem(
                    prescription_id=prescription.id,
                    sort_order=index,
                    **item.model_dump(),
                )
            )
        await self.db.flush()

        await self.transition(visit, "completed")
        return await self.get_final_prescription(visit_id)

    async def get_final_prescription(self, visit_id: UUID) -> Optional[FinalPrescription]:
        result = await self.db.execute(
            select(FinalPrescription)
            .options(selectinload(FinalPrescription.items))
            .where(FinalPrescription.visit_id == visit_id)
        )
        return result.scalar_one_or_none()

    # =====================================================================
    # DOCTOR <-> PATIENT CHAT
    # =====================================================================

    async def add_chat_message(
        self, visit_id: UUID, sender_id: UUID, sender_role: str, content: str
    ) -> ChatMessage:
        message = ChatMessage(
            visit_id=visit_id,
            sender_id=sender_id,
            sender_role=sender_role,
            content=content,
            sent_at=datetime.utcnow(),
        )
        self.db.add(message)
        await self.db.flush()
        return message

    async def list_chat_messages(self, visit_id: UUID) -> list[ChatMessage]:
        result = await self.db.execute(
            select(ChatMessage)
            .where(ChatMessage.visit_id == visit_id)
            .order_by(ChatMessage.sent_at.asc(), ChatMessage.created_at.asc())
        )
        return result.scalars().all()
