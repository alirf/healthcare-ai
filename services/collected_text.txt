# app/middleware/jwt_validation.py
import jwt
from fastapi import Request
from fastapi.responses import JSONResponse
from pathlib import Path
from starlette.middleware.base import BaseHTTPMiddleware

from app.config import settings


PUBLIC_ROUTES = {
    "/auth/login",
    "/auth/register",
    "/auth/verify-otp",
    "/auth/refresh",
    "/auth/health",
    "/api/auth/login",
    "/api/auth/register", 
    "/api/auth/verify-otp",
    "/api/auth/refresh",
    "/api/auth/health",
    "/docs",
    "/openapi.json",
    "/redoc",
    "/health",
}


class JWTValidationMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, **kwargs):
        super().__init__(app)
        self.public_key = Path(settings.PUBLIC_KEY_PATH).read_text()
    
    async def dispatch(self, request: Request, call_next):
        path = request.url.path
        
        if any(path.startswith(route) for route in PUBLIC_ROUTES):
            return await call_next(request)
        
        auth_header = request.headers.get("authorization")
        if not auth_header:
            return JSONResponse(
                status_code=401,
                content={"detail": "Token not sent.", "code": "MISSING_TOKEN"}
            )
        
        if not auth_header.startswith("Bearer "):
            return JSONResponse(
                status_code=401,
                content={"detail": "The token format is invalid.", "code": "INVALID_TOKEN_FORMAT"}
            )
        
        token = auth_header.replace("Bearer ", "")
        
        try:
            payload = jwt.decode(
                token,
                self.public_key,
                algorithms=["RS256"],
                options={"require": ["exp", "sub", "role"]}
            )
            
            request.state.user_id = payload.get("sub")
            request.state.user_role = payload.get("role")
            request.state.token_payload = payload
            
        except jwt.ExpiredSignatureError:
            return JSONResponse(
                status_code=401,
                content={"detail": "Token has expired.", "code": "TOKEN_EXPIRED"}
            )
        except jwt.InvalidTokenError as e:
            return JSONResponse(
                status_code=401,
                content={"detail": f"Token is invalid: {str(e)}", "code": "INVALID_TOKEN"}
            )
        
        return await call_next(request)

import time
import json
from fastapi import Request
from starlette.middleware.base import BaseHTTPMiddleware


class LoggingMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        start_time = time.time()
        
        # اطلاعات درخواست
        method = request.method
        path = request.url.path
        query = str(request.query_params)
        client_ip = request.client.host if request.client else "unknown"
        user_agent = request.headers.get("user-agent", "unknown")
        user_id = getattr(request.state, "user_id", "anonymous")
        user_role = getattr(request.state, "user_role", "none")
        
        # ارسال درخواست
        try:
            response = await call_next(request)
            status_code = response.status_code
        except Exception as exc:
            status_code = 500
            raise exc
        finally:
            process_time = (time.time() - start_time) * 1000  # میلی‌ثانیه
            
            log_entry = {
                "timestamp": time.strftime("%Y-%m-%dT%H:%M:%S"),
                "method": method,
                "path": path,
                "query": query if query else None,
                "status_code": status_code,
                "process_time_ms": round(process_time, 2),
                "client_ip": client_ip,
                "user_id": user_id,
                "user_role": user_role,
                "user_agent": user_agent,
            }
            
            # لاگ در stdout (در Docker با json formatter قابل جمع‌آوری است)
            print(json.dumps(log_entry, ensure_ascii=False))
        
        # اضافه کردن هدر زمان پاسخ
        response.headers["X-Process-Time"] = str(round(process_time, 2))
        return response

import time
from fastapi import Request
from fastapi.responses import JSONResponse
from starlette.middleware.base import BaseHTTPMiddleware

from app.config import settings
from app.utils.redis_client import get_redis_client


class RateLimiterMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, **kwargs):
        super().__init__(app)
        self.redis = get_redis_client()
        self.public_limit = settings.RATE_LIMIT_PUBLIC
        self.auth_limit = settings.RATE_LIMIT_AUTH
        self.ai_triage_limit = settings.RATE_LIMIT_AI_TRIAGE
        self.profile_limit = settings.RATE_LIMIT_PROFILE
    
    def _get_key(self, identifier: str, window: int) -> str:
        current_window = int(time.time()) // window
        return f"rate_limit:{identifier}:{current_window}"
    
    def _check_limit(self, key: str, limit: int, window: int = 60) -> tuple[bool, int, int]:
        pipe = self.redis.pipeline()
        pipe.incr(key)
        pipe.expire(key, window)
        results = pipe.execute()
        
        current_count = results[0]
        remaining = max(0, limit - current_count)
        reset_after = window - (int(time.time()) % window)
        
        allowed = current_count <= limit
        return allowed, remaining, reset_after
    
    async def dispatch(self, request: Request, call_next):
        path = request.url.path
        
        public_paths = ["/api/auth/login", "/api/auth/register", "/api/auth/verify-otp"]
        if any(path.startswith(p) for p in public_paths):
            identifier = f"ip:{request.client.host}"
            allowed, remaining, reset_after = self._check_limit(
                self._get_key(identifier, 60), self.public_limit
            )
            if not allowed:
                return JSONResponse(
                    status_code=429,
                    content={
                        "detail": "Access from this IP address has been rate-limited",
                        "code": "RATE_LIMIT_EXCEEDED",
                        "retry_after": reset_after
                    },
                    headers={"Retry-After": str(reset_after)}
                )
            response = await call_next(request)
            response.headers["X-RateLimit-Remaining"] = str(remaining)
            return response
        
        profile_paths = ["/api/profile/"]
        if any(path.startswith(p) for p in profile_paths):
            user_id = getattr(request.state, "user_id", None)
            if user_id:
                identifier = f"user:{user_id}:profile"
                allowed, remaining, reset_after = self._check_limit(
                    self._get_key(identifier, 60), self.profile_limit
                )
                if not allowed:
                    return JSONResponse(
                        status_code=429,
                        content={
                            "detail": "Access from this IP address has been rate-limited",
                            "code": "PROFILE_RATE_LIMIT_EXCEEDED",
                            "retry_after": reset_after
                        },
                        headers={"Retry-After": str(reset_after)}
                    )
                response = await call_next(request)
                response.headers["X-RateLimit-Remaining"] = str(remaining)
                return response
        
        if path.startswith("/api/intake/") and "ai" in path:
            user_id = getattr(request.state, "user_id", None)
            if user_id:
                identifier = f"user:{user_id}:ai"
                allowed, remaining, reset_after = self._check_limit(
                    self._get_key(identifier, 60), self.ai_triage_limit
                )
                if not allowed:
                    return JSONResponse(
                        status_code=429,
                        content={
                            "detail": "Access from this IP address has been rate-limited for AI Triage",
                            "code": "AI_RATE_LIMIT_EXCEEDED",
                            "retry_after": reset_after
                        },
                        headers={"Retry-After": str(reset_after)}
                    )
                response = await call_next(request)
                response.headers["X-RateLimit-Remaining"] = str(remaining)
                return response
        
        user_id = getattr(request.state, "user_id", None)
        if user_id:
            identifier = f"user:{user_id}"
            allowed, remaining, reset_after = self._check_limit(
                self._get_key(identifier, 60), self.auth_limit
            )
            if not allowed:
                return JSONResponse(
                    status_code=429,
                    content={
                        "detail": "Access from this IP address has been rate-limited",
                        "code": "RATE_LIMIT_EXCEEDED",
                        "retry_after": reset_after
                    },
                    headers={"Retry-After": str(reset_after)}
                )
            response = await call_next(request)
            response.headers["X-RateLimit-Remaining"] = str(remaining)
            return response
        
        return await call_next(request)

from fastapi import Request
from fastapi.responses import JSONResponse
from starlette.middleware.base import BaseHTTPMiddleware


PROTECTED_ROUTES = {
    "/api/doctor-review/": ["DOCTOR", "ADMIN"],
    "/api/prescriptions/create": ["DOCTOR", "ADMIN"],
    "/api/prescriptions/approve": ["DOCTOR", "ADMIN"],
    "/api/admin/": ["ADMIN"],
    "/api/profile/patient/me": ["PATIENT"],
    "/api/profile/patient/me/allergies": ["PATIENT"],
    "/api/profile/patient/me/medications": ["PATIENT"],
    "/api/profile/patient/me/conditions": ["PATIENT"],
    "/api/profile/patient/me/avatar": ["PATIENT"],
    "/api/profile/doctor/me": ["DOCTOR"],
    "/api/profile/doctor/me/avatar": ["DOCTOR"],
}


class RoleCheckerMiddleware(BaseHTTPMiddleware):
    def __init__(self, app, **kwargs):
        super().__init__(app)
    
    async def dispatch(self, request: Request, call_next):
        path = request.url.path
        user_role = getattr(request.state, "user_role", None)
        
        for route_prefix, allowed_roles in PROTECTED_ROUTES.items():
            if path.startswith(route_prefix):
                if user_role not in allowed_roles:
                    return JSONResponse(
                        status_code=403,
                        content={
                            "detail": "Forbidden: You do not have the required role to access this resource",
                            "code": "FORBIDDEN",
                            "required_roles": allowed_roles,
                            "your_role": user_role
                        }
                    )
                break
        
        return await call_next(request)

# app/router/proxy.py
import httpx
from fastapi import Request, APIRouter, HTTPException
from fastapi.responses import JSONResponse

from app.config import settings


router = APIRouter()

SERVICES = {
    "auth": settings.AUTH_SERVICE_URL,
    "profile": settings.PROFILE_SERVICE_URL,
}

# ✅ مسیرهای هر سرویس
SERVICE_PATHS = {
    "auth": "/auth",
    "profile": "/api/profile",
}

INTERNAL_ONLY = {"triage"}


async def proxy_request(service: str, path: str, request: Request):
    """پروکسی درخواست به سرویس مقصد"""
    
    if service in INTERNAL_ONLY:
        raise HTTPException(status_code=403, detail="Forbidden: Internal service only")
    
    if service not in SERVICES:
        raise HTTPException(status_code=404, detail=f"Service '{service}' not found")
    
    # ✅ مسیر درست بر اساس سرویس
    base_path = SERVICE_PATHS.get(service, f"/api/{service}")
    target_url = f"{SERVICES[service]}{base_path}/{path}"
    
    headers = dict(request.headers)
    headers.pop("host", None)
    
    user_id = getattr(request.state, "user_id", None)
    user_role = getattr(request.state, "user_role", None)
    
    if user_id:
        headers["x-user-id"] = str(user_id)
    if user_role:
        headers["x-user-role"] = str(user_role)
    
    headers["x-gateway-request"] = "true"
    
    if path.startswith("internal/"):
        headers["x-internal-api-key"] = settings.INTERNAL_API_KEY
    
    body = await request.body()
    
    async with httpx.AsyncClient(timeout=30.0) as client:
        try:
            response = await client.request(
                method=request.method,
                url=target_url,
                headers=headers,
                params=request.query_params,
                content=body if body else None,
            )
            
            content_type = response.headers.get("content-type", "")
            if "application/json" in content_type:
                response_content = response.json()
            else:
                response_content = {
                    "detail": response.text or "Empty response from upstream",
                    "upstream_status": response.status_code
                }
            
            gateway_response = JSONResponse(
                content=response_content,
                status_code=response.status_code,
            )
            
            # کپی کردن هدرهای مهم از upstream
            for header in ["content-type", "x-request-id"]:
                if header in response.headers:
                    gateway_response.headers[header] = response.headers[header]
            
            return gateway_response
            
        except httpx.ConnectError:
            raise HTTPException(status_code=503, detail=f"Service '{service}' unavailable")
        except httpx.TimeoutException:
            raise HTTPException(status_code=504, detail=f"Timeout connecting to '{service}'")

from fastapi import Depends, HTTPException, status
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from sqlalchemy.orm import Session
from app.database import get_db
from app.core.security import decode_token
from app.models import User

security = HTTPBearer()

def get_current_user(
    credentials: HTTPAuthorizationCredentials = Depends(security),
    db: Session = Depends(get_db)
) -> User:
    """استخراج کاربر فعلی از توکن"""
    try:
        payload = decode_token(credentials.credentials)
        if payload.get("type") != "access":
            raise HTTPException(status_code=401, detail="توکن نامعتبر است")
        
        user_id = payload.get("sub")
        user = db.query(User).filter(User.id == user_id).first()
        
        if not user or not user.is_active:
            raise HTTPException(status_code=401, detail="کاربر یافت نشد")
        
        return user
        
    except Exception as e:
        raise HTTPException(status_code=401, detail="توکن نامعتبر یا منقضی شده")

def get_current_doctor(current_user: User = Depends(get_current_user)) -> User:
    """بررسی نقش پزشک"""
    if current_user.role != "DOCTOR":
        raise HTTPException(status_code=403, detail="دسترسی محدود به پزشکان")
    return current_user

def get_current_admin(current_user: User = Depends(get_current_user)) -> User:
    """بررسی نقش ادمین"""
    if current_user.role != "ADMIN":
        raise HTTPException(status_code=403, detail="دسترسی محدود به ادمین")
    return current_user

    def register(self, data: UserRegisterRequest) -> User:
        existing = self.db.query(User).filter(User.phone_number == data.phone_number).first()
        if existing:
            raise UserExistsException()

        if data.email:
            existing_email = self.db.query(User).filter(User.email == data.email).first()
            if existing_email:
                from app.core.exceptions import AuthException
                raise AuthException("این ایمیل قبلاً ثبت شده است", 409)
        
        user = User(
            id=uuid.uuid4(),
            phone_number=data.phone_number,
            email=data.email,
            password_hash=hash_password(data.password),
            role=data.role,
            is_phone_verified=False
        )
        self.db.add(user)
        self.db.commit()
        self.db.refresh(user)
        return user
    
    def create_otp(self, phone_number: str, purpose: str, user_id: uuid.UUID = None) -> str:
        code = generate_otp()
        
        rate_key = f"otp_rate:{phone_number}"
        if not check_rate_limit(rate_key, limit=3, window=600):
            raise RateLimitException()
        
        self.db.query(OTPCode).filter(
            OTPCode.phone_number == phone_number,
            OTPCode.purpose == purpose,
            OTPCode.is_used == False
        ).update({"is_used": True})
        
        otp = OTPCode(
            id=uuid.uuid4(),
            user_id=user_id,
            phone_number=phone_number,
            code_hash=hash_otp(code),
            purpose=purpose,
            expires_at=get_otp_expiry()
        )
        self.db.add(otp)
        self.db.commit()
        
        print(f"📱 OTP برای {phone_number}: {code}")
        return code
    
    def verify_otp_code(self, data: OTPVerifyRequest, purpose: str = "REGISTER") -> User:
        otp = self.db.query(OTPCode).filter(
            OTPCode.phone_number == data.phone_number,
            OTPCode.is_used == False,
            OTPCode.purpose == purpose
        ).order_by(OTPCode.created_at.desc()).first()
        
        if not otp:
            raise OTPInvalidException()
        
        now = datetime.now(timezone.utc)
        expires_at = otp.expires_at
        if expires_at.tzinfo is None:
            expires_at = expires_at.replace(tzinfo=timezone.utc)
        
        if expires_at < now:
            raise OTPExpiredException()
        
        if not verify_otp(data.code, otp.code_hash):
            raise OTPInvalidException()
        
        user = self.db.query(User).filter(User.phone_number == data.phone_number).first()
        if not user:
            raise InvalidCredentialsException()
        
        if purpose == "REGISTER":
            user.is_phone_verified = True
        
        otp.is_used = True
        self.db.commit()
        self.db.refresh(user)
        
        return user
    
    def login(self, data: LoginRequest, ip_address: str = None) -> dict:
        rate_key = f"login_rate:{data.phone_number}"
        if not check_rate_limit(rate_key, limit=5, window=900):
            raise RateLimitException()
        
        user = self.db.query(User).filter(User.phone_number == data.phone_number).first()
        
        attempt = LoginAttempt(
            id=uuid.uuid4(),
            phone_number=data.phone_number,
            ip_address=ip_address,
            success=False
        )
        self.db.add(attempt)
        
        if not user or not verify_password(data.password, user.password_hash):
            self.db.commit()
            raise InvalidCredentialsException()
        
        if not user.is_active:
            self.db.commit()
            raise InvalidCredentialsException()
        
        if not user.is_phone_verified:
            self.db.commit()
            raise InvalidCredentialsException()
        
        attempt.success = True
        self.db.commit()
        
        return self._generate_tokens(user)
    
    def login_with_otp(self, data: OTPVerifyRequest) -> dict:
        user = self.verify_otp_code(data, purpose="LOGIN")
        return self._generate_tokens(user)
    
    def refresh_access_token(self, data: RefreshTokenRequest) -> dict:
        try:
            payload = decode_token(data.refresh_token)
            if payload.get("type") != "refresh":
                raise InvalidCredentialsException()
            
            user_id = payload.get("sub")
            user = self.db.query(User).filter(User.id == user_id).first()
            
            if not user or not user.is_active:
                raise InvalidCredentialsException()
            
            jti = payload.get("jti") or str(uuid.uuid4())
            if is_token_blacklisted(jti):
                raise InvalidCredentialsException()
            
            token_hash = hash_password(data.refresh_token)
            rt = self.db.query(RefreshToken).filter(
                RefreshToken.user_id == user_id,
                RefreshToken.token_hash == token_hash,
                RefreshToken.revoked == False,
                RefreshToken.expires_at > datetime.now(timezone.utc)
            ).first()
            
            if not rt:
                raise InvalidCredentialsException()
            
            rt.revoked = True
            self.db.commit()
            
            return self._generate_tokens(user)
            
        except Exception:
            raise InvalidCredentialsException()
    
    def logout(self, user_id: str, refresh_token: str = None):
        if refresh_token:
            try:
                payload = decode_token(refresh_token)
                jti = payload.get("jti") or str(uuid.uuid4())
                exp = payload.get("exp", 0)
                now = int(datetime.now(timezone.utc).timestamp())
                ttl = max(exp - now, 60)
                blacklist_token(jti, ttl)
                
                token_hash = hash_password(refresh_token)
                self.db.query(RefreshToken).filter(
                    RefreshToken.token_hash == token_hash
                ).update({"revoked": True})
                self.db.commit()
            except Exception:
                pass
        
        return {"message": "با موفقیت خارج شدید"}
    
    def reset_password(self, data: PasswordResetRequest) -> dict:
        user = self.verify_otp_code(
            OTPVerifyRequest(phone_number=data.phone_number, code=data.code),
            purpose="RESET_PASSWORD"
        )
        
        user.password_hash = hash_password(data.new_password)
        self.db.commit()
        
        self.db.query(RefreshToken).filter(
            RefreshToken.user_id == user.id
        ).update({"revoked": True})
        self.db.commit()
        
        return {"message": "رمز عبور با موفقیت تغییر کرد"}
    
    def _generate_tokens(self, user: User) -> dict:
        access_token = create_access_token(user.id, user.role, user.is_phone_verified)
        refresh_token = create_refresh_token(user.id)
        
        rt = RefreshToken(
            id=uuid.uuid4(),
            user_id=user.id,
            token_hash=hash_password(refresh_token),
            expires_at=datetime.now(timezone.utc) + timedelta(days=7)
        )
        self.db.add(rt)
        self.db.commit()
        
        return {
            "access_token": access_token,
            "refresh_token": refresh_token,
            "expires_in": 900,
            "token_type": "bearer",
            "user": {
                "id": str(user.id),
                "phone_number": user.phone_number,
                "role": user.role,
                "is_phone_verified": user.is_phone_verified
            }
        }
    
    def get_user_by_id(self, user_id: uuid.UUID) -> User:
        return self.db.query(User).filter(User.id == user_id).first()

import redis
from app.config import settings

redis_client = redis.Redis(
    host=settings.REDIS_HOST,
    port=settings.REDIS_PORT,
    decode_responses=True
)

def get_redis():
    return redis_client

def set_rate_limit(key: str, limit: int, window: int):
    """تنظیم محدودیت نرخ"""
    current = redis_client.get(key)
    if current is None:
        redis_client.setex(key, window, 1)
        return True, limit - 1
    current = int(current)
    if current >= limit:
        return False, 0
    redis_client.incr(key)
    return True, limit - current - 1

def check_rate_limit(key: str, limit: int = 5, window: int = 900) -> bool:
    """بررسی محدودیت نرخ (پیش‌فرض: 5 بار در 15 دقیقه)"""
    allowed, remaining = set_rate_limit(key, limit, window)
    return allowed

def blacklist_token(token_jti: str, expiry: int):
    """افزودن توکن به لیست سیاه"""
    redis_client.setex(f"blacklist:{token_jti}", expiry, "1")

def is_token_blacklisted(token_jti: str) -> bool:
    """بررسی لیست سیاه"""
    return redis_client.exists(f"blacklist:{token_jti}") == 1

def cache_otp(phone: str, code: str, expiry: int = 300):
    """ذخیره OTP در Redis"""
    redis_client.setex(f"otp:{phone}", expiry, code)

def get_cached_otp(phone: str) -> str:
    """دریافت OTP از Redis"""
    return redis_client.get(f"otp:{phone}")

def delete_cached_otp(phone: str):
    """حذف OTP از Redis"""
    redis_client.delete(f"otp:{phone}")

pwd_context = CryptContext(schemes=["argon2"], deprecated="auto")

def hash_password(password: str) -> str:
    return pwd_context.hash(password)

def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)

def load_private_key() -> str:
    with open(settings.JWT_PRIVATE_KEY_PATH, "r") as f:
        return f.read()

def load_public_key() -> str:
    with open(settings.JWT_PUBLIC_KEY_PATH, "r") as f:
        return f.read()

def create_access_token(user_id: str, role: str, phone_verified: bool) -> str:
    private_key = load_private_key()
    now = datetime.now(timezone.utc)
    payload = {
        "sub": str(user_id),
        "role": role,
        "phone_verified": phone_verified,
        "iat": now,
        "exp": now + timedelta(seconds=settings.JWT_ACCESS_EXPIRATION),
        "type": "access",
        "iss": settings.APP_NAME
    }
    return jwt.encode(payload, private_key, algorithm=settings.JWT_ALGORITHM)

def create_refresh_token(user_id: str) -> str:
    private_key = load_private_key()
    now = datetime.now(timezone.utc)
    payload = {
        "sub": str(user_id),
        "iat": now,
        "exp": now + timedelta(seconds=settings.JWT_REFRESH_EXPIRATION),
        "type": "refresh",
        "iss": settings.APP_NAME
    }
    return jwt.encode(payload, private_key, algorithm=settings.JWT_ALGORITHM)

def decode_token(token: str) -> dict:
    public_key = load_public_key()
    return jwt.decode(token, public_key, algorithms=[settings.JWT_ALGORITHM], issuer=settings.APP_NAME)

"""Authentication dependencies for FastAPI."""
from typing import Optional

from fastapi import Depends, HTTPException, status, Request
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import JWTError, jwt

from app.config import get_settings
from app.exceptions import InternalAPIKeyError

security = HTTPBearer(auto_error=False)


async def get_current_user(
    request: Request,
    credentials: Optional[HTTPAuthorizationCredentials] = Depends(security)
) -> dict:
    """Validate user identity from JWT OR trusted Gateway headers.
    
    Priority:
    1. If x-gateway-request header exists → Trust x-user-id/x-user-role (Gateway already validated)
    2. Otherwise → Validate JWT locally (direct access or other services)
    """
    settings = get_settings()
    
    # ➕ NEW: Check if request came from Gateway (trusted)
    is_gateway_request = request.headers.get("x-gateway-request") == "true"
    user_id_from_gateway = request.headers.get("x-user-id")
    user_role_from_gateway = request.headers.get("x-user-role")
    
    if is_gateway_request and user_id_from_gateway:
        # Gateway already validated JWT, trust the headers
        return {
            "sub": user_id_from_gateway,
            "role": user_role_from_gateway,
            "source": "gateway"
        }
    
    # ➕ Fallback: Validate JWT locally (direct access, testing, etc.)
    if not credentials:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authorization header missing",
            headers={"WWW-Authenticate": "Bearer"},
        )
    
    try:
        with open(settings.JWT_PUBLIC_KEY_PATH, "r") as f:
            public_key = f.read()
        
        payload = jwt.decode(
            credentials.credentials,
            public_key,
            algorithms=[settings.JWT_ALGORITHM],
        )
        
        user_id: str = payload.get("sub")
        if user_id is None:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid token: missing subject",
            )
        
        payload["source"] = "jwt"
        return payload
        
    except JWTError as e:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=f"Invalid token: {str(e)}",
            headers={"WWW-Authenticate": "Bearer"},
        )
    except FileNotFoundError:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="JWT public key not found",
        )


def require_role(required_role: str):
    """Dependency factory to enforce role-based access control."""
    async def role_checker(
        current_user: dict = Depends(get_current_user)
    ) -> dict:
        user_roles = current_user.get("roles", [])
        # Handle both list and single role
        if isinstance(user_roles, str):
            user_roles = [user_roles]
        user_role = current_user.get("role")
        if user_role:
            user_roles.append(user_role)
        
        if required_role not in user_roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Access denied. Required role: {required_role}",
            )
        return current_user
    
    return role_checker


async def verify_internal_api_key(
    request: Request,
    x_internal_api_key: Optional[str] = None
) -> bool:
    """Verify service-to-service API key for internal endpoints."""
    settings = get_settings()
    
    if x_internal_api_key is None:
        x_internal_api_key = request.headers.get("X-Internal-API-Key")
    
    if not x_internal_api_key or x_internal_api_key != settings.INTERNAL_API_KEY:
        raise InternalAPIKeyError()
    
    return True

"""FastAPI application factory for User Profile Service."""
from contextlib import asynccontextmanager

from fastapi import FastAPI, Request
from fastapi.middleware.cors import CORSMiddleware
from fastapi.responses import JSONResponse

from app.config import get_settings
from app.routers import patient, doctor, internal

settings = get_settings()


@asynccontextmanager
async def lifespan(app: FastAPI):
    """Application lifespan handler for startup/shutdown events."""
    # Startup
    print(f"🚀 {settings.APP_NAME} starting up...")
    yield
    # Shutdown
    print(f"🛑 {settings.APP_NAME} shutting down...")


def create_application() -> FastAPI:
    """Create and configure the FastAPI application."""
    app = FastAPI(
        title=settings.APP_NAME,
        description="""
        Healthcare AI - User Profile Service
        
        Manages patient and doctor profiles including:
        - Basic demographic information
        - Allergies and reactions
        - Current medications
        - Chronic conditions
        - Profile avatars (stored in object storage)
        
        ## Authentication
        All patient/doctor endpoints require a valid JWT Bearer token.
        Internal endpoints require an `X-Internal-API-Key` header.
        
        ## HIPAA Compliance
        All access to Protected Health Information (PHI) is logged
        in the audit log for compliance purposes.
        """,
        version="1.0.0",
        lifespan=lifespan,
        docs_url="/docs" if settings.DEBUG else None,
        redoc_url="/redoc" if settings.DEBUG else None,
    )
    
    # CORS middleware
    app.add_middleware(
        CORSMiddleware,
        allow_origins=["*"] if settings.DEBUG else [],
        allow_credentials=True,
        allow_methods=["*"],
        allow_headers=["*"],
    )
    
    # Include routers
    app.include_router(patient.router)
    app.include_router(doctor.router)
    app.include_router(internal.router)
    
    # Global exception handler
    @app.exception_handler(Exception)
    async def global_exception_handler(request: Request, exc: Exception):
        """Handle uncaught exceptions gracefully."""
        return JSONResponse(
            status_code=500,
            content={"detail": "Internal server error"},
        )
    
    # Health check endpoint
    @app.get("/health", tags=["Health"])
    async def health_check():
        """Health check endpoint for load balancers and monitoring."""
        return {"status": "healthy", "service": settings.APP_NAME}
    
    # Root endpoint
    @app.get("/", tags=["Root"])
    async def root():
        """Root endpoint with service information."""
        return {
            "service": settings.APP_NAME,
            "version": "1.0.0",
            "docs": "/docs" if settings.DEBUG else None,
        }
    
    return app


# Create the application instance
app = create_application()

@router.post("/me/avatar")
async def upload_doctor_avatar(
    request: Request,
    file: UploadFile = File(...),
    current_user: dict = Depends(require_role("doctor")),
    db: AsyncSession = Depends(get_db),
):
    """Upload doctor profile picture."""
    storage = StorageService()
    service = ProfileService(db)
    audit = AuditService(db)
    
    user_id = UUID(current_user["sub"])
    
    image_url, object_key = await storage.upload_avatar(
        file=file,
        user_id=str(user_id),
        user_type="doctor",
    )
    
    old_profile = await service.get_doctor_profile(user_id)
    if old_profile and old_profile.profile_image_key:
        await storage.delete_file(old_profile.profile_image_key)
    
    # Note: need to add update_doctor_avatar to ProfileService
    # For now using direct update
    result = await db.execute(
        select(DoctorProfile).where(DoctorProfile.user_id == user_id)
    )
    profile = result.scalar_one_or_none()
    if profile:
        profile.profile_image_url = image_url
        profile.profile_image_key = object_key
        await db.flush()
    
    ip, ua = get_client_info(request)
    await audit.log_update(
        user_id=user_id,
        table_name="doctor_profiles",
        record_id=user_id,
        old_values={"profile_image_url": old_profile.profile_image_url if old_profile else None},
        new_values={"profile_image_url": image_url},
        performed_by=user_id,
        ip_address=ip,
        user_agent=ua,
    )
    
    return {"profile_image_url": image_url}

"""Authentication dependencies for FastAPI."""
from typing import Optional

from fastapi import Depends, HTTPException, status, Request
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import JWTError, jwt

from app.config import get_settings
from app.exceptions import InternalAPIKeyError

security = HTTPBearer(auto_error=False)


async def get_current_user(
    credentials: Optional[HTTPAuthorizationCredentials] = Depends(security)
) -> dict:
    """Validate JWT token and return user payload.
    
    This dependency is used for patient-facing endpoints.
    The JWT is signed by the Auth Service using RS256.
    """
    if not credentials:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail="Authorization header missing",
            headers={"WWW-Authenticate": "Bearer"},
        )
    
    settings = get_settings()
    
    try:
        # Read public key from file
        with open(settings.JWT_PUBLIC_KEY_PATH, "r") as f:
            public_key = f.read()
        
        payload = jwt.decode(
            credentials.credentials,
            public_key,
            algorithms=[settings.JWT_ALGORITHM],
        )
        
        user_id: str = payload.get("sub")
        if user_id is None:
            raise HTTPException(
                status_code=status.HTTP_401_UNAUTHORIZED,
                detail="Invalid token: missing subject",
            )
        
        return payload
        
    except JWTError as e:
        raise HTTPException(
            status_code=status.HTTP_401_UNAUTHORIZED,
            detail=f"Invalid token: {str(e)}",
            headers={"WWW-Authenticate": "Bearer"},
        )
    except FileNotFoundError:
        raise HTTPException(
            status_code=status.HTTP_500_INTERNAL_SERVER_ERROR,
            detail="JWT public key not found",
        )


def require_role(required_role: str):
    """Dependency factory to enforce role-based access control.
    
    Usage:
        @router.get("/me")
        async def get_profile(current_user: dict = Depends(require_role("patient"))):
            ...
    """
    async def role_checker(
        current_user: dict = Depends(get_current_user)
    ) -> dict:
        user_roles = current_user.get("roles", [])
        if required_role not in user_roles:
            raise HTTPException(
                status_code=status.HTTP_403_FORBIDDEN,
                detail=f"Access denied. Required role: {required_role}",
            )
        return current_user
    
    return role_checker


async def verify_internal_api_key(
    request: Request,
    x_internal_api_key: Optional[str] = None
) -> bool:
    """Verify service-to-service API key for internal endpoints.
    
    Internal endpoints should NOT be exposed through the API Gateway.
    They are only accessible directly between microservices.
    """
    settings = get_settings()
    
    # Get from header if not passed directly
    if x_internal_api_key is None:
        x_internal_api_key = request.headers.get("X-Internal-API-Key")
    
    if not x_internal_api_key or x_internal_api_key != settings.INTERNAL_API_KEY:
        raise InternalAPIKeyError()
    
    return True

class AuditService:
    """Service for logging all access to protected health information."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
    
    async def log_action(
        self,
        user_id: UUID,
        action: str,
        table_name: str,
        record_id: Optional[UUID] = None,
        old_values: Optional[dict] = None,
        new_values: Optional[dict] = None,
        performed_by: Optional[UUID] = None,
        ip_address: Optional[str] = None,
        user_agent: Optional[str] = None,
    ) -> ProfileAuditLog:
        """Log an action on protected health information.
        
        Args:
            user_id: The patient/doctor whose data was accessed
            action: One of: create, update, delete, view
            table_name: The database table affected
            record_id: The specific record ID
            old_values: Previous values (for updates/deletes)
            new_values: New values (for creates/updates)
            performed_by: User who performed the action (defaults to user_id)
            ip_address: Client IP address
            user_agent: Client user agent string

            The created audit log entry
        """
        log = ProfileAuditLog(
            user_id=user_id,
            action=action,
            table_name=table_name,
            record_id=record_id,
            old_values=old_values,
            new_values=new_values,
            performed_by=performed_by or user_id,
            ip_address=ip_address,
            user_agent=user_agent,

class ProfileService:
    """Service layer for profile operations."""
    
    def __init__(self, db: AsyncSession):
        self.db = db
    
    # =====================================================================
    # PATIENT PROFILE OPERATIONS
    # =====================================================================
    
    async def get_patient_profile(self, user_id: UUID) -> Optional[PatientProfile]:
        """Get patient profile with all related data eagerly loaded."""
        result = await self.db.execute(
            select(PatientProfile)
            .options(
                selectinload(PatientProfile.allergies),
                selectinload(PatientProfile.medications),
                selectinload(PatientProfile.conditions),
            )
            .where(PatientProfile.user_id == user_id)
        )
        return result.scalar_one_or_none()
    
    async def patient_profile_exists(self, user_id: UUID) -> bool:
        """Check if a patient profile exists."""
        result = await self.db.execute(
            select(exists().where(PatientProfile.user_id == user_id))
        )
        return result.scalar()
    
    async def create_patient_profile(
        self, user_id: UUID, data: PatientProfileCreate
    ) -> PatientProfile:
        """Create a new patient profile."""
        # Check if profile already exists
        if await self.patient_profile_exists(user_id):
            raise ProfileAlreadyExistsError()
        
        profile = PatientProfile(
            user_id=user_id,
            **data.model_dump(),
        )
        self.db.add(profile)
        await self.db.flush()
        return profile
    
    async def update_patient_profile(
        self, user_id: UUID, data: PatientProfileUpdate
    ) -> Optional[PatientProfile]:
        """Update patient profile fields. Only updates provided fields."""
        result = await self.db.execute(
            select(PatientProfile).where(PatientProfile.user_id == user_id)
        )
        profile = result.scalar_one_or_none()
        if not profile:
            return None
        
        update_data = data.model_dump(exclude_unset=True)
        for field, value in update_data.items():
            setattr(profile, field, value)
        
        await self.db.flush()
        return profile
    
    async def delete_patient_profile(self, user_id: UUID) -> bool:
        """Delete a patient profile and all related records (cascade)."""
        result = await self.db.execute(
            select(PatientProfile).where(PatientProfile.user_id == user_id)
        )
        profile = result.scalar_one_or_none()
        if not profile:
            return False
        
        await self.db.delete(profile)
        await self.db.flush()

    async def get_medical_summary(self, user_id: UUID) -> Optional[MedicalSummaryResponse]:
        """Generate condensed medical summary for AI services.
        
        This is called automatically at the start of each consultation
        so the patient does not have to re-enter known information.
        """
        profile = await self.get_patient_profile(user_id)
        if not profile:
            return None
        
        # Calculate age from birth_date
        today = date.today()
        age = today.year - profile.birth_date.year
        if (today.month, today.day) < (profile.birth_date.month, profile.birth_date.day):
            age -= 1
        
        # Calculate BMI
        bmi = None
        if profile.height_cm and profile.weight_kg:
            height_m = profile.height_cm / 100
            bmi = round(profile.weight_kg / (height_m ** 2), 1)
        
        # Filter active records only
        active_medications = [m for m in profile.medications if m.is_active]
        active_conditions = [c for c in profile.conditions if c.is_active]
        
        return MedicalSummaryResponse(
            user_id=profile.user_id,
            full_name=profile.full_name,
            age=age,
            gender=profile.gender or "unknown",
            bmi=bmi,
            blood_type=profile.blood_type,
            allergies=[
                {"name": a.allergy_name, "severity": a.severity}
                for a in profile.allergies
            ],
            current_medications=[
                {
                    "name": m.medication_name,
                    "dosage": m.dosage,
                    "frequency": m.frequency,
                    "prescribed_by": m.prescribed_by,
                }
                for m in active_medications
            ],
            chronic_conditions=[
                {
                    "name": c.condition_name,
                    "severity": c.severity,
                    "icd10_code": c.icd10_code,
                }
                for c in active_conditions
            ],
        )

"""Object storage service for file uploads (MinIO/S3-compatible)."""
import uuid
from typing import Optional

import boto3
from botocore.client import Config
from botocore.exceptions import ClientError
from fastapi import UploadFile

from app.config import get_settings
from app.exceptions import InvalidFileError, FileTooLargeError


class StorageService:
    """Service for uploading and managing files in object storage."""
    
    def __init__(self):
        settings = get_settings()
        self.settings = settings
        
        protocol = "https" if settings.MINIO_SECURE else "http"
        endpoint_url = f"{protocol}://{settings.MINIO_ENDPOINT}"
        
        self.client = boto3.client(
            "s3",
            endpoint_url=endpoint_url,
            aws_access_key_id=settings.MINIO_ACCESS_KEY,
            aws_secret_access_key=settings.MINIO_SECRET_KEY,
            config=Config(signature_version="s3v4"),
            region_name="us-east-1",
        )
        self.bucket = settings.MINIO_BUCKET_NAME
        self._ensure_bucket_exists()
    
    def _ensure_bucket_exists(self):
        """Create bucket if it does not exist."""
        try:
            self.client.head_bucket(Bucket=self.bucket)
        except ClientError as e:
            error_code = e.response["Error"]["Code"]
            if error_code == "404":
                self.client.create_bucket(Bucket=self.bucket)
            else:
                raise
    
    def validate_file(self, file: UploadFile) -> None:
        """Validate uploaded file type and size."""
        # Check file type
        if file.content_type not in self.settings.ALLOWED_IMAGE_TYPES:
            raise InvalidFileError(
                f"Invalid file type: {file.content_type}. "
                f"Allowed types: {', '.join(self.settings.ALLOWED_IMAGE_TYPES)}"
            )
    
    async def upload_avatar(
        self, file: UploadFile, user_id: str, user_type: str = "patient"
    ) -> tuple[str, str]:
        """Upload an avatar image to object storage.
        
        Args:
            file: The uploaded file
            user_id: UUID of the user
            user_type: "patient" or "doctor"
            
        Returns:
            Tuple of (public_url, object_key)
        """
        self.validate_file(file)
        
        # Generate unique object key
        ext = file.filename.split(".")[-1] if "." in file.filename else "jpg"
        object_key = f"avatars/{user_type}s/{user_id}/{uuid.uuid4()}.{ext}"
        
        # Read file content
        content = await file.read()
        
        # Check file size
        max_size = self.settings.MAX_AVATAR_SIZE_MB * 1024 * 1024
        if len(content) > max_size:
            raise FileTooLargeError(self.settings.MAX_AVATAR_SIZE_MB)
        
        # Upload to storage
        self.client.put_object(
            Bucket=self.bucket,
            Key=object_key,
            Body=content,
            ContentType=file.content_type,
            Metadata={
                "uploaded-by": "profile-service",
                "user-id": user_id,
                "user-type": user_type,
            },
        )
        
        # Generate presigned URL (valid for 7 days)
        url = self.client.generate_presigned_url(
            "get_object",
            Params={"Bucket": self.bucket, "Key": object_key},
            ExpiresIn=3600 * 24 * 7,
        )
        
        return url, object_key
    
    async def delete_file(self, object_key: str) -> bool:
        """Delete a file from object storage.
        
        Returns:
            True if deleted successfully, False if not found.
        """
        try:
            self.client.delete_object(Bucket=self.bucket, Key=object_key)
            return True
        except ClientError:
            return False
    
    async def get_file_url(self, object_key: str, expires_in: int = 3600) -> Optional[str]:
        """Generate a presigned URL for accessing a file.
        
        Args:
            object_key: The storage object key
            expires_in: URL expiration time in seconds
            
        Returns:
            Presigned URL or None if object not found
        """
        try:
            url = self.client.generate_presigned_url(
                "get_object",
                Params={"Bucket": self.bucket, "Key": object_key},
                ExpiresIn=expires_in,
            )
            return url
        except ClientError:
            return None

"""Utility helper functions."""
from datetime import date
from uuid import UUID


def calculate_age(birth_date: date) -> int:
    """Calculate age from birth date."""
    today = date.today()
    age = today.year - birth_date.year
    if (today.month, today.day) < (birth_date.month, birth_date.day):
        age -= 1
    return age


def calculate_bmi(height_cm: int, weight_kg: int) -> float:
    """Calculate BMI from height (cm) and weight (kg)."""
    height_m = height_cm / 100
    return round(weight_kg / (height_m ** 2), 1)


def is_valid_uuid(value: str) -> bool:
    """Check if a string is a valid UUID."""
    try:
        UUID(value)
        return True
    except ValueError:
        return False
